     1→"use client";
     2→
     3→import { useState, useEffect } from "react";
     4→import { useTranslations } from "next-intl";
     5→import { Link, usePathname } from "@/i18n/navigation";
     6→import { Menu, X } from "lucide-react";
     7→import { Container } from "@/components/ui/container";
     8→import { buttonClassName } from "@/components/ui/button-styles";
     9→import { LanguageSwitch } from "./language-switch";
    10→import { ThemeToggle } from "./theme-toggle";
    11→import { cn } from "@/lib/utils";
    12→
    13→const navItems = [
    14→  { href: "/services", key: "services" },
    15→  { href: "/work", key: "work" },
    16→  { href: "/offers", key: "offers" },
    17→  { href: "/about", key: "about" },
    18→  { href: "/contact", key: "contact" },
    19→] as const;
    20→
    21→export function SiteHeader() {
    22→  const t = useTranslations("Nav");
    23→  const common = useTranslations("Common");
    24→  const pathname = usePathname();
    25→  const [isScrolled, setIsScrolled] = useState(false);
    26→  const [isMobileMenuOpen, setIsMobileMenuOpen] = useState(false);
    27→
    28→  // Handle scroll effect
    29→  useEffect(() => {
    30→    const handleScroll = () => {
    31→      setIsScrolled(window.scrollY > 20);
    32→    };
    33→
    34→    window.addEventListener("scroll", handleScroll, { passive: true });
    35→    return () => window.removeEventListener("scroll", handleScroll);
    36→  }, []);
    37→
    38→  // Close mobile menu on route change
    39→  useEffect(() => {
    40→    setIsMobileMenuOpen(false);
    41→  }, [pathname]);
    42→
    43→  // Prevent body scroll when mobile menu is open
    44→  useEffect(() => {
    45→    if (isMobileMenuOpen) {
    46→      document.body.style.overflow = "hidden";
    47→    } else {
    48→      document.body.style.overflow = "";
    49→    }
    50→    return () => {
    51→      document.body.style.overflow = "";
    52→    };
    53→  }, [isMobileMenuOpen]);
    54→
    55→  return (
    56→    <>
    57→      <header
    58→        className={cn(
    59→          "sticky top-0 z-50 transition-all duration-300",
    60→          isScrolled
    61→            ? "border-b border-border bg-surface/80 shadow-sm backdrop-blur-xl"
    62→            : "bg-transparent"
    63→        )}
    64→      >
    65→        <Container>
    66→          <div className="flex h-[var(--header-height)] items-center justify-between gap-4">
    67→            {/* Logo */}
    68→            <Link href="/" className="group flex items-center gap-3">
    69→              <span className="flex h-10 w-10 items-center justify-center rounded-[var(--radius-md)] border border-border bg-surface text-sm font-bold text-primary transition-all duration-300 group-hover:border-primary/40 group-hover:shadow-glow">
    70→                Y
    71→              </span>
    72→              <div className="leading-tight">
    73→                <p className="text-base font-semibold text-text">Yodev</p>
    74→                <p className="text-xs text-muted">{common("tagline")}</p>
    75→              </div>
    76→            </Link>
    77→
    78→            {/* Desktop Navigation */}
    79→            <nav className="hidden items-center gap-1 md:flex">
    80→              {navItems.map((item) => {
    81→                const isActive = pathname === item.href;
    82→                return (
    83→                  <Link
    84→                    key={item.href}
    85→                    href={item.href}
    86→                    className={cn(
    87→                      "relative px-4 py-2 text-sm font-medium transition-colors duration-200",
    88→                      isActive
    89→                        ? "text-primary"
    90→                        : "text-muted hover:text-text"
    91→                    )}
    92→                  >
    93→                    {t(item.key)}
    94→                    {isActive && (
    95→                      <span className="absolute bottom-0 left-4 right-4 h-0.5 rounded-full bg-primary" />
    96→                    )}
    97→                  </Link>
    98→                );
    99→              })}
   100→            </nav>
   101→
   102→            {/* Desktop Actions */}
   103→            <div className="hidden items-center gap-3 md:flex">
   104→              <ThemeToggle />
   105→              <LanguageSwitch />
   106→              <Link
   107→                href="/contact"
   108→                className={buttonClassName({ size: "sm" })}
   109→              >
   110→                {common("cta")}
   111→              </Link>
   112→            </div>
   113→
   114→            {/* Mobile Actions */}
   115→            <div className="flex items-center gap-2 md:hidden">
   116→              <ThemeToggle />
   117→              <LanguageSwitch />
   118→              <button
   119→                onClick={() => setIsMobileMenuOpen(!isMobileMenuOpen)}
   120→                className="flex h-10 w-10 items-center justify-center rounded-[var(--radius-md)] border border-border bg-surface text-text transition-colors hover:border-primary/40"
   121→                aria-label={isMobileMenuOpen ? common("menuClose") : common("menuOpen")}
   122→              >
   123→                {isMobileMenuOpen ? (
   124→                  <X className="h-5 w-5" />
   125→                ) : (
   126→                  <Menu className="h-5 w-5" />
   127→                )}
   128→              </button>
   129→            </div>
   130→          </div>
   131→        </Container>
   132→      </header>
   133→
   134→      {/* Mobile Menu Overlay */}
   135→      <div
   136→        className={cn(
   137→          "fixed inset-0 z-40 bg-bg/80 backdrop-blur-sm transition-opacity duration-300 md:hidden",
   138→          isMobileMenuOpen
   139→            ? "opacity-100"
   140→            : "pointer-events-none opacity-0"
   141→        )}
   142→        onClick={() => setIsMobileMenuOpen(false)}
   143→      />
   144→
   145→      {/* Mobile Menu Panel */}
   146→      <div
   147→        className={cn(
   148→          "fixed right-0 top-0 z-50 h-full w-[280px] border-l border-border bg-surface shadow-xl transition-transform duration-300 ease-out md:hidden",
   149→          isMobileMenuOpen ? "translate-x-0" : "translate-x-full"
   150→        )}
   151→      >
   152→        <div className="flex h-[var(--header-height)] items-center justify-between border-b border-border px-6">
   153→          <span className="text-sm font-semibold text-text">{common("menu")}</span>
   154→          <button
   155→            onClick={() => setIsMobileMenuOpen(false)}
   156→            className="flex h-9 w-9 items-center justify-center rounded-[var(--radius-sm)] text-muted transition-colors hover:bg-bg-2 hover:text-text"
   157→            aria-label={common("menuClose")}
   158→          >
   159→            <X className="h-5 w-5" />
   160→          </button>
   161→        </div>
   162→
   163→        <nav className="flex flex-col p-6">
   164→          {navItems.map((item) => {
   165→            const isActive = pathname === item.href;
   166→            return (
   167→              <Link
   168→                key={item.href}
   169→                href={item.href}
   170→                className={cn(
   171→                  "border-b border-border py-4 text-base font-medium transition-colors",
   172→                  isActive
   173→                    ? "text-primary"
   174→                    : "text-text hover:text-primary"
   175→                )}
   176→              >
   177→                {t(item.key)}
   178→              </Link>
   179→            );
   180→          })}
   181→        </nav>
   182→
   183→        <div className="absolute bottom-0 left-0 right-0 border-t border-border p-6">
   184→          <Link
   185→            href="/contact"
   186→            className={buttonClassName({ size: "md", className: "w-full justify-center" })}
   187→          >
   188→            {common("cta")}
   189→          </Link>
   190→        </div>
   191→      </div>
   192→    </>
   193→  );
   194→}
