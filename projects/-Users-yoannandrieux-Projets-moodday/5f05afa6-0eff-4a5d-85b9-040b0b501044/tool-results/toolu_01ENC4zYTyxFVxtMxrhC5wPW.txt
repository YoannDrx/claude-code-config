     9→async function calculateTotalMRR() {
    10→  const subscriptions = await prisma.subscription.findMany({
    11→    where: {
    12→      status: { in: ["active", "trialing", "past_due"] },
    13→      stripeSubscriptionId: { not: null },
    14→    },
    15→    select: {
    16→      stripeSubscriptionId: true,
    17→    },
    18→  });
    19→
    20→  if (subscriptions.length === 0) {
    21→    return 0;
    22→  }
    23→
    24→  const stripeSubscriptions = (
    25→    await Promise.all(
    26→      subscriptions.map(async (sub) => {
    27→        if (!sub.stripeSubscriptionId) return null;
    28→
    29→        try {
    30→          const stripeSub = await stripe.subscriptions.retrieve(
    31→            sub.stripeSubscriptionId,
    32→          );
    33→          return stripeSub;
    34→        } catch {
    35→          return null;
    36→        }
    37→      }),
    38→    )
    39→  ).filter((sub) => sub !== null);
    40→
    41→  let totalMRR = 0;
    42→
    43→  for (const stripeSub of stripeSubscriptions) {
    44→    for (const item of stripeSub.items.data) {
    45→      const price = item.price;
    46→      if (!price.unit_amount || !price.recurring) continue;
    47→
    48→      const { unit_amount } = price;
    49→      const { interval, interval_count } = price.recurring;
    50→
    51→      let monthlyAmount = 0;
    52→      switch (interval) {
    53→        case "month":
    54→          monthlyAmount = unit_amount / interval_count;
    55→          break;
    56→        case "year":
    57→          monthlyAmount = unit_amount / (12 * interval_count);
    58→          break;
    59→        case "week":
    60→          monthlyAmount = (unit_amount * 4.33) / interval_count;
    61→          break;
    62→        case "day":
    63→          monthlyAmount = (unit_amount * 30) / interval_count;
    64→          break;
    65→      }
    66→
    67→      totalMRR += monthlyAmount * (item.quantity ?? 1);
    68→    }
    69→  }
    70→
    71→  return totalMRR;
    72→}
