    20→  const invoices = await stripe.invoices.list({
    21→    created: {
    22→      gte: Math.floor(sixMonthsAgo.getTime() / 1000),

    31→  for (let i = 0; i < 6; i++) {
    32→    const date = new Date(now);
    33→    date.setMonth(date.getMonth() - i);
    34→    const key = `${date.getFullYear()}-${String(date.getMonth() + 1).padStart(2, "0")}`;
    35→    monthlyMrr[key] = 0;
    36→  }
    37→
    38→  for (const invoice of invoices.data) {
    39→    if (!invoice.amount_paid || !invoice.created) continue;
    40→
    41→    const date = new Date(invoice.created * 1000);
    42→    const key = `${date.getFullYear()}-${String(date.getMonth() + 1).padStart(2, "0")}`;
    43→
    44→    if (key in monthlyMrr) {
    45→      monthlyMrr[key] += invoice.amount_paid;

    47→  }

    51→      date,
