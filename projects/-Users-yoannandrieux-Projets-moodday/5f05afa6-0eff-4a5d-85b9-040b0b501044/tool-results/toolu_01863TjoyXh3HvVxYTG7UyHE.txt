     1→"use server";
     2→
     3→import { authAction } from "@/lib/actions/safe-actions";
     4→import { AUTH_PLANS } from "@/lib/auth/stripe/auth-plans";
     5→import { ActionError } from "@/lib/errors/action-error";
     6→import { prisma } from "@/lib/prisma";
     7→import { getServerUrl } from "@/lib/server-url";
     8→import { stripe } from "@/lib/stripe";
     9→import { z } from "zod";
    10→
    11→export const upgradeUserAction = authAction
    12→  .inputSchema(
    13→    z.object({
    14→      plan: z.string(),
    15→      annual: z.boolean().default(false),
    16→      successUrl: z.string(),
    17→      cancelUrl: z.string(),
    18→    }),
    19→  )
    20→  .action(
    21→    async ({
    22→      parsedInput: { plan, annual, successUrl, cancelUrl },
    23→      ctx: { user },
    24→    }) => {
    25→      // Find the plan
    26→      const authPlan = AUTH_PLANS.find((p) => p.name === plan);
    27→      if (!authPlan) {
    28→        throw new ActionError(`Plan "${plan}" not found`);
    29→      }
    30→
    31→      // Get the price ID based on annual or monthly
    32→      const priceId = annual
    33→        ? authPlan.annualDiscountPriceId
    34→        : authPlan.priceId;
    35→      if (!priceId) {
    36→        throw new ActionError(`Price ID not found for plan "${plan}"`);
    37→      }
    38→
    39→      // Get the full user from database to access stripeCustomerId
    40→      const dbUser = await prisma.user.findUnique({
    41→        where: { id: user.id },
    42→        select: { stripeCustomerId: true },
    43→      });
    44→
    45→      if (!dbUser?.stripeCustomerId) {
    46→        throw new ActionError("No Stripe customer ID found");
    47→      }
    48→
    49→      const customerId = dbUser.stripeCustomerId;
    50→
    51→      // Create checkout session
    52→      const session = await stripe.checkout.sessions.create({
    53→        customer: customerId,
    54→        payment_method_types: ["card"],
    55→        line_items: [
    56→          {
    57→            price: priceId,
    58→            quantity: 1,
    59→          },
    60→        ],
    61→        mode: "subscription",
    62→        success_url: `${getServerUrl()}${successUrl}?session_id={CHECKOUT_SESSION_ID}`,
    63→        cancel_url: `${getServerUrl()}${cancelUrl}`,
    64→        metadata: {
    65→          userId: user.id,
    66→          plan: plan,
    67→        },
    68→        subscription_data: {
    69→          metadata: {
    70→            userId: user.id,
    71→            plan: plan,
    72→          },
    73→          trial_period_days: authPlan.freeTrial?.days,
    74→        },
    75→      });
    76→
    77→      if (!session.url) {
    78→        throw new ActionError("Failed to create checkout session");
    79→      }
    80→
    81→      return {
    82→        url: session.url,
    83→      };
    84→    },
    85→  );
