     1→"use server";
     2→
     3→import { authAction } from "@/lib/actions/safe-actions";
     4→import { ActionError } from "@/lib/errors/action-error";
     5→import { prisma } from "@/lib/prisma";
     6→import { getServerUrl } from "@/lib/server-url";
     7→import { stripe } from "@/lib/stripe";
     8→import { z } from "zod";
     9→
    10→const getStripeCustomerId = async (userId: string) => {
    11→  const user = await prisma.user.findUnique({
    12→    where: { id: userId },
    13→    select: { stripeCustomerId: true },
    14→  });
    15→
    16→  if (!user?.stripeCustomerId) {
    17→    throw new ActionError("No stripe customer id found");
    18→  }
    19→
    20→  return user.stripeCustomerId;
    21→};
    22→
    23→export const openStripePortalAction = authAction.action(
    24→  async ({ ctx: { user } }) => {
    25→    const stripeCustomerId = await getStripeCustomerId(user.id);
    26→
    27→    if (!stripeCustomerId) {
    28→      throw new ActionError("No stripe customer id found");
    29→    }
    30→
    31→    const stripeBilling = await stripe.billingPortal.sessions.create({
    32→      customer: stripeCustomerId,
    33→      return_url: `${getServerUrl()}/account/billing`,
    34→    });
    35→
    36→    if (!stripeBilling.url) {
    37→      throw new ActionError("Failed to create stripe billing portal session");
    38→    }
    39→
    40→    return {
    41→      url: stripeBilling.url,
    42→    };
    43→  },
    44→);
    45→
    46→export const cancelSubscriptionAction = authAction
    47→  .inputSchema(
    48→    z.object({
    49→      returnUrl: z.string().url(),
    50→    }),
    51→  )
    52→  .action(async ({ parsedInput: { returnUrl }, ctx: { user } }) => {
    53→    const stripeCustomerId = await getStripeCustomerId(user.id);
    54→
    55→    if (!stripeCustomerId) {
    56→      throw new ActionError("No stripe customer id found");
    57→    }
    58→
    59→    // Get the current subscription
    60→    const subscription = await prisma.subscription.findFirst({
    61→      where: { referenceId: user.id },
    62→    });
    63→
    64→    if (!subscription?.stripeSubscriptionId) {
    65→      throw new ActionError("No active subscription found");
    66→    }
    67→
    68→    // Create billing portal session which allows the user to cancel
    69→    const stripeBilling = await stripe.billingPortal.sessions.create({
    70→      customer: stripeCustomerId,
    71→      return_url: `${getServerUrl()}${returnUrl}`,
    72→    });
    73→
    74→    if (!stripeBilling.url) {
    75→      throw new ActionError("Failed to create stripe billing portal session");
    76→    }
    77→
    78→    return {
    79→      url: stripeBilling.url,
    80→    };
    81→  });
