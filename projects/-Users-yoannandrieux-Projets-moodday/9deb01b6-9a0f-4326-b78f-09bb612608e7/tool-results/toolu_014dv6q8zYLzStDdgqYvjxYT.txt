     1→"use client";
     2→
     3→import { Avatar, AvatarFallback } from "@/components/ui/avatar";
     4→import {
     5→  Card,
     6→  CardDescription,
     7→  CardFooter,
     8→  CardHeader,
     9→} from "@/components/ui/card";
    10→import {
    11→  Form,
    12→  FormControl,
    13→  FormField,
    14→  FormItem,
    15→  FormLabel,
    16→  FormMessage,
    17→  useZodForm,
    18→} from "@/components/ui/form";
    19→import { Input } from "@/components/ui/input";
    20→import { LoadingButton } from "@/features/form/submit-button";
    21→import { useI18n } from "@/i18n/provider";
    22→import { authClient } from "@/lib/auth-client";
    23→import { unwrapSafePromise } from "@/lib/promises";
    24→import { useMutation } from "@tanstack/react-query";
    25→import { Lock } from "lucide-react";
    26→import { useRouter } from "next/navigation";
    27→import { toast } from "sonner";
    28→import { z } from "zod";
    29→
    30→const EmailFormSchema = z.object({
    31→  email: z.string().email(),
    32→});
    33→
    34→export function ForgetPasswordPage() {
    35→  const router = useRouter();
    36→  const { t } = useI18n();
    37→
    38→  const emailForm = useZodForm({
    39→    schema: EmailFormSchema,
    40→  });
    41→
    42→  const forgetPasswordMutation = useMutation({
    43→    mutationFn: async (values: { email: string }) => {
    44→      return unwrapSafePromise(
    45→        authClient.requestPasswordReset({
    46→          email: values.email,
    47→          redirectTo: "/auth/reset-password",
    48→        }),
    49→      );
    50→    },
    51→    onError: (error) => {
    52→      toast.error(error.message);
    53→    },
    54→    onSuccess: () => {
    55→      router.push("/auth/verify");
    56→    },
    57→  });
    58→
    59→  function onSubmitEmail(values: z.infer<typeof EmailFormSchema>) {
    60→    forgetPasswordMutation.mutate(values);
    61→  }
    62→
    63→  return (
    64→    <Card className="mx-auto w-full max-w-md lg:max-w-lg lg:p-6">
    65→      <CardHeader>
    66→        <div className="flex justify-center">
    67→          <Avatar className="size-16">
    68→            <AvatarFallback>
    69→              <Lock />
    70→            </AvatarFallback>
    71→          </Avatar>
    72→        </div>
    73→        <CardHeader className="text-center">
    74→          {t("auth.forgetPassword.title")}
    75→        </CardHeader>
    76→
    77→        <CardDescription className="text-center">
    78→          {t("auth.forgetPassword.description")}
    79→        </CardDescription>
    80→      </CardHeader>
    81→
    82→      <CardFooter className="border-t pt-6">
    83→        <Form
    84→          form={emailForm}
    85→          onSubmit={onSubmitEmail}
    86→          className="w-full space-y-4"
    87→        >
    88→          <FormField
    89→            control={emailForm.control}
    90→            name="email"
    91→            render={({ field }) => (
    92→              <FormItem>
    93→                <FormLabel>{t("auth.form.email")}</FormLabel>
    94→                <FormControl>
    95→                  <Input
    96→                    placeholder={t("auth.form.emailPlaceholder")}
    97→                    {...field}
    98→                  />
    99→                </FormControl>
   100→                <FormMessage />
   101→              </FormItem>
   102→            )}
   103→          />
   104→          <LoadingButton
   105→            loading={forgetPasswordMutation.isPending}
   106→            type="submit"
   107→            className="w-full"
   108→          >
   109→            {t("auth.forgetPassword.submit")}
   110→          </LoadingButton>
   111→        </Form>
   112→      </CardFooter>
   113→    </Card>
   114→  );
   115→}
