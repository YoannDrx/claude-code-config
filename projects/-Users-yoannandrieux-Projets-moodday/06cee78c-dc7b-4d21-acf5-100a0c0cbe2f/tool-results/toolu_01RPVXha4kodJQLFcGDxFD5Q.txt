    10→/**
    11→ * Helper function to create a test account
    12→ * @returns Object containing the test user's credentials
    13→ */
    14→export async function createTestAccount(options: {
    15→  page: Page;
    16→  callbackURL?: string;
    17→  initialUserData?: { name: string; email: string; password: string };
    18→  admin?: boolean;
    19→}) {
    20→  // Generate fake user data
    21→  const userData = options.initialUserData ?? {
    22→    name: faker.person.fullName(),
    23→    email: getUserEmail(),
    24→    password: faker.internet.password({ length: 12, memorable: true }),
    25→  };
    26→
    27→  // Navigate to signup page
    28→  await options.page.goto(`/auth/signup?callbackUrl=${options.callbackURL}`);
    29→
    30→  // Fill out the form (supports both English and French labels)
    31→  await options.page.getByLabel(/^Name$|^Nom$/i).fill(userData.name);
    32→  await options.page.getByLabel(/^Email$/i).fill(userData.email);
    33→  await options.page.locator('input[name="password"]').fill(userData.password);
    34→  await options.page
    35→    .locator('input[name="verifyPassword"]')
    36→    .fill(userData.password);
    37→
    38→  // Submit the form (supports both English "Create account" and French "Créer un compte")
    39→  await options.page
    40→    .getByRole("button", { name: /create account|créer un compte/i })
    41→    .click();
    42→
    43→  // Wait for navigation to complete - we should be redirected to the callback URL
    44→  if (options.callbackURL) {
    45→    await options.page.waitForLoadState("networkidle");
    46→    await options.page.waitForURL(new RegExp(options.callbackURL), {
    47→      timeout: 30000,
    48→    });
    49→  }
    50→
    51→  if (options.admin) {
    52→    const user = await retry(
    53→      async () =>
    54→        prisma.user.findUniqueOrThrow({
    55→          where: { email: userData.email },
    56→        }),
    57→      {
    58→        maxAttempts: 5,
    59→        delayMs: 1000,
    60→        backoff: true,
    61→      },
    62→    );
    63→    logger.info("Creating admin user", user);
    64→    await prisma.user.update({
    65→      where: { id: user.id },
    66→      data: { role: "admin" },
    67→    });
    68→    // await 5 seconds
    69→    await new Promise((resolve) => setTimeout(resolve, 5000));
    70→  }
    71→
    72→  return userData;
    73→}

    93→  await page.locator('input[name="password"]').fill(userData.password);
    94→
    95→  // Submit the form (supports both English "Sign in" and French "Se connecter")

    99→    .click();
   100→
   101→  // Wait for navigation to complete if a callback URL is provided
