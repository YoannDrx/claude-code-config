     1→"use client";
     2→
     3→import {
     4→  Collapsible,
     5→  CollapsibleContent,
     6→  CollapsibleTrigger,
     7→} from "@/components/ui/collapsible";
     8→import {
     9→  Sidebar,
    10→  SidebarContent,
    11→  SidebarFooter,
    12→  SidebarGroup,
    13→  SidebarGroupContent,
    14→  SidebarGroupLabel,
    15→  SidebarHeader,
    16→  SidebarRail,
    17→} from "@/components/ui/sidebar";
    18→import { SidebarNavigationMenu } from "@/components/ui/sidebar-utils";
    19→import type { NavigationGroup } from "@/features/navigation/navigation.type";
    20→import { SidebarUserButton } from "@/features/sidebar/sidebar-user-button";
    21→import { useI18n } from "@/i18n/provider";
    22→import { ChevronDown } from "lucide-react";
    23→import { usePathname } from "next/navigation";
    24→import type { PropsWithChildren } from "react";

    27→
    28→export function AdminSidebar() {
    29→  const { t } = useI18n();
    30→  const links: NavigationGroup[] = getAdminNavigation(t);
    31→
    32→  return (
    33→    <Sidebar variant="inset">
    34→      <SidebarHeader className="flex flex-col gap-2">
    35→        <div className="flex items-center gap-2 px-2 py-1.5">
    36→          <div className="bg-primary text-primary-foreground flex size-8 items-center justify-center rounded-lg">
    37→            <span className="text-sm font-semibold">A</span>
    38→          </div>
    39→          <span className="font-semibold">{t("admin.panel")}</span>
    40→        </div>
    41→      </SidebarHeader>
    42→      <SidebarContent>
    43→        {links.map((link) => (
    44→          <ItemCollapsing
    45→            defaultOpenStartPath={link.defaultOpenStartPath}
    46→            key={link.title}
    47→          >
    48→            <SidebarGroup key={link.title}>
    49→              <SidebarGroupLabel asChild>
    50→                <CollapsibleTrigger>
    51→                  {link.title}
    52→                  <ChevronDown className="ml-auto transition-transform group-data-[state=open]/collapsible:rotate-180" />
    53→                </CollapsibleTrigger>
    54→              </SidebarGroupLabel>
    55→              <CollapsibleContent>
    56→                <SidebarGroupContent>
    57→                  <SidebarNavigationMenu link={link} />
    58→                </SidebarGroupContent>
    59→              </CollapsibleContent>
    60→            </SidebarGroup>
    61→          </ItemCollapsing>
    62→        ))}
    63→      </SidebarContent>
    64→      <SidebarFooter className="flex flex-col gap-2">
    65→        <SidebarUserButton />
    66→      </SidebarFooter>
    67→      <SidebarRail />
    68→    </Sidebar>
    69→  );
    70→}
    71→
    72→const ItemCollapsing = (
    73→  props: PropsWithChildren<{ defaultOpenStartPath?: string }>,
    74→) => {
    75→  const [open, setOpen] = useState(false);
    76→  const pathname = usePathname();
    77→
    78→  const isOpen = props.defaultOpenStartPath
    79→    ? pathname.startsWith(props.defaultOpenStartPath)
    80→    : true;
    81→
    82→  useEffect(() => {
    83→    if (isOpen) {
    84→      setOpen(isOpen);
    85→    }
    86→  }, [isOpen]);
    87→  return (
    88→    <Collapsible
    89→      defaultOpen={isOpen}
    90→      onOpenChange={setOpen}
    91→      open={open}
    92→      className="group/collapsible"
    93→    >
    94→      {props.children}
    95→    </Collapsible>
    96→  );
    97→};
