     1→import type { Statements } from "better-auth/plugins/access";

     9→const b2cStatements = {

    12→  comparison: ["create", "read", "update", "delete"], // Compare properties
    13→  alert: ["create", "read", "update", "delete"], // Property alerts
    14→  savedSearch: ["create", "read", "update", "delete"], // Saved search criteria
    15→
    16→  // Financial tools

    18→  budget: ["create", "read", "update"], // Budget calculations
    19→
    20→  // Profile management
    21→  profile: ["read", "update"], // User profile
    22→  preferences: ["read", "update"], // Search preferences
    23→  notifications: ["read", "update"], // Notification settings
    24→} as const satisfies Statements;
    25→
    26→export const B2CPermissionSchema = z.object(
    27→  Object.fromEntries(
    28→    Object.entries(b2cStatements).map(([key, actions]) => [
    29→      key,
    30→      z
    31→        .array(z.enum([...actions] as unknown as [string, ...string[]]))
    32→        .optional(),
    33→    ]),
    34→  ) as unknown as {
    35→    [K in keyof typeof b2cStatements]: z.ZodOptional<
    36→      z.ZodArray<
    37→        z.ZodEnum<
    38→          [
    39→            ...((typeof b2cStatements)[K] extends readonly (infer T)[]
    40→              ? T extends string
    41→                ? (typeof b2cStatements)[K] extends readonly [
    42→                    string,
    43→                    ...string[],
    44→                  ]
    45→                  ? (typeof b2cStatements)[K]
    46→                  : never
    47→                : never
    48→              : never),
    49→          ]
    50→        >
    51→      >
    52→    >;
    53→  },
    54→);
    55→
    56→export type B2CPermission = z.infer<typeof B2CPermissionSchema>;
    57→
    58→// Create access control for B2C
    59→export const b2cAc = createAccessControl(b2cStatements);
    60→
    61→// Define B2C user role (individual buyers)
    62→const individualBuyer = b2cAc.newRole({
    63→  // Full access to personal features

    65→  comparison: ["create", "read", "update", "delete"],
    66→  alert: ["create", "read", "update", "delete"],
    67→  savedSearch: ["create", "read", "update", "delete"],
    68→  simulation: ["create", "read", "update", "delete"],
    69→  budget: ["create", "read", "update"],

    71→  preferences: ["read", "update"],
    72→  notifications: ["read", "update"],

    74→
    75→export const b2cRoles = { individualBuyer } as const;
    76→
    77→export const B2CRolesKeys = ["individualBuyer"] as const;
    78→
    79→export type B2CRole = keyof typeof b2cRoles;
    80→
    81→// Helper functions for B2C permissions
    82→export const hasB2CPermission = (
    83→  userRole: B2CRole | undefined,
    84→  resource: keyof typeof b2cStatements,
    85→  action: string,
    86→): boolean => {
    87→  if (!userRole) {
    88→    return false;
    89→  }
    90→
    91→  const role = b2cRoles[userRole];
    92→  const resourcePermissions = role.statements[resource];
    93→
    94→  return resourcePermissions.includes(action as never);
    95→};
    96→
    97→export const getB2CUserRole = (accountType: string): B2CRole | null => {
    98→  if (accountType === "individual") {
    99→    return "individualBuyer";
   100→  }
   101→  return null;
   102→};
