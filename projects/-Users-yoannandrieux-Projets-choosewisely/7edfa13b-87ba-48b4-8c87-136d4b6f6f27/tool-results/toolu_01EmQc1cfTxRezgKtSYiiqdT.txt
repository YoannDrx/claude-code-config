     1→import { createSafeActionClient } from "next-safe-action";
     2→import { z } from "zod";
     3→import { AuthPermissionSchema, RolesKeys } from "../auth/auth-permissions";
     4→import { getRequiredUser } from "../auth/auth-user";
     5→import { logger } from "../logger";
     6→import { getRequiredCurrentOrg } from "../organizations/get-org";
     7→
     8→export class ActionError extends Error {}
     9→
    10→type handleServerError = (e: Error) => string;
    11→
    12→const handleServerError: handleServerError = (e) => {
    13→  if (e instanceof ActionError) {
    14→    logger.debug("[DEV] - Action Error", e.message);
    15→    return e.message;
    16→  }
    17→
    18→  logger.info("Unknown Error", e);
    19→
    20→  if (process.env.NODE_ENV === "development") {
    21→    return e.message;
    22→  }
    23→
    24→  return "An unexpected error occurred.";
    25→};
    26→
    27→export const action = createSafeActionClient({
    28→  handleServerError,
    29→});
    30→
    31→export const authAction = createSafeActionClient({
    32→  handleServerError,
    33→}).use(async ({ next }) => {
    34→  const user = await getRequiredUser();
    35→
    36→  return next({
    37→    ctx: {
    38→      user: user,
    39→    },
    40→  });
    41→});
    42→
    43→export const orgAction = createSafeActionClient({
    44→  handleServerError,
    45→  defineMetadataSchema() {
    46→    return z
    47→      .object({
    48→        roles: z.array(z.enum(RolesKeys)).optional(),
    49→        permissions: AuthPermissionSchema.optional(),
    50→      })
    51→      .optional();
    52→  },
    53→}).use(async ({ next, metadata = {} }) => {
    54→  try {
    55→    const org = await getRequiredCurrentOrg(metadata);
    56→    return next({
    57→      ctx: { org },
    58→    });
    59→  } catch {
    60→    throw new ActionError(
    61→      "You need to be part of an organization to access this resource.",
    62→    );
    63→  }
    64→});
