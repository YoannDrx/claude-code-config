     1→import { SiteConfig } from "@/site-config";

     6→  id: string;
     7→  email: string;
     8→  accountType?: string;

    10→
    11→/**
    12→ * Get user data from session for middleware
    13→ * Note: Limited to cookie data only, no database queries allowed in middleware
    14→ */
    15→export async function getMiddlewareUser(request: NextRequest) {
    16→  try {
    17→    const session = getSessionCookie(request, {
    18→      cookiePrefix: SiteConfig.appId,
    19→    }) as { user: SessionUser } | string | null;
    20→
    21→    if (!session || typeof session === "string") {
    22→      return null;
    23→    }
    24→
    25→    // Return basic user data from cookie session
    26→    // For full user data, make API call from page/component instead
    27→    return {
    28→      id: session.user.id,
    29→      email: session.user.email,
    30→      // Default to individual if not specified in cookie
    31→      accountType: session.user.accountType ?? "individual",
    32→    };
    33→  } catch {
    34→    return null;
    35→  }
    36→}
