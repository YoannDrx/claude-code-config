     1→import { getMiddlewareUser, getRedirectUrl } from "@/lib/auth/middleware-utils";
     2→import { SiteConfig } from "@/site-config";
     3→import { getSessionCookie } from "better-auth/cookies";
     4→import type { NextRequest } from "next/server";
     5→import { NextResponse } from "next/server";
     6→
     7→export async function middleware(request: NextRequest) {
     8→  const { pathname } = request.nextUrl;
     9→
    10→  // Redirect homepage based on account type
    11→  if (pathname === "/") {
    12→    const session = getSessionCookie(request, {
    13→      cookiePrefix: SiteConfig.appId,
    14→    });
    15→
    16→    if (session) {
    17→      const user = await getMiddlewareUser(request);
    18→      if (user) {
    19→        const url = new URL(request.url);
    20→        url.pathname = getRedirectUrl(user.accountType);
    21→        return NextResponse.redirect(url.toString());
    22→      }
    23→    }
    24→  }
    25→
    26→  // B2C routes protection - redirect professionals to /orgs
    27→  if (
    28→    pathname.startsWith("/dashboard") ||
    29→    pathname.startsWith("/favorites") ||
    30→    pathname.startsWith("/comparisons") ||
    31→    pathname.startsWith("/alerts")
    32→  ) {
    33→    const user = await getMiddlewareUser(request);
    34→    if (user?.accountType === "professional") {
    35→      const url = new URL(request.url);
    36→      url.pathname = "/orgs";
    37→      return NextResponse.redirect(url.toString());
    38→    }
    39→  }
    40→
    41→  // B2B routes protection - redirect individuals to /dashboard
    42→  if (pathname.startsWith("/orgs")) {
    43→    const user = await getMiddlewareUser(request);
    44→    if (user?.accountType === "individual") {
    45→      const url = new URL(request.url);
    46→      url.pathname = "/dashboard";
    47→      return NextResponse.redirect(url.toString());
    48→    }
    49→  }
    50→
    51→  return NextResponse.next();
    52→}
    53→
    54→export const config = {
    55→  matcher: [
    56→    /*
    57→     * Match all request paths except for the ones starting with:
    58→     * - api (API routes)
    59→     * - _next/static (static files)
    60→     * - _next/image (image optimization files)
    61→     * - favicon.ico, sitemap.xml, robots.txt (metadata files)
    62→     */
    63→    "/((?!api|_next/static|_next/image|favicon.ico|sitemap.xml|robots.txt).*)",
    64→  ],
    65→};
