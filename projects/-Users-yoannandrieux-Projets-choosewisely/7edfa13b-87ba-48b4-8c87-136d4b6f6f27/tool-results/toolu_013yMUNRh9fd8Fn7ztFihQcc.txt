    26→export const hasUnifiedPermission = async (
    27→  permission: AuthPermission | B2BPermission,
    28→) => {
    29→  const userContext = await getUserContext();
    30→
    31→  if (!userContext) {
    32→    return false;
    33→  }
    34→
    35→  // For professionals, use B2B permissions if it's a B2B-specific permission
    36→  if (userContext.accountType === "professional") {
    37→    try {
    38→      // Try B2B permissions first
    39→      return await hasB2BPermissionAsync(permission as B2BPermission);
    40→    } catch {
    41→      // Fall back to legacy organization permissions
    42→      return hasPermission(permission as AuthPermission);
    43→    }
    44→  }
    45→
    46→  // For individuals, B2B permissions don't apply
    47→  return false;
    48→};
