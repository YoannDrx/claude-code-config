    59→  constructor(config: MoteurImmoClientConfig) {
    60→    this.apiKey = config.apiKey;
    61→    this.baseUrl = config.baseUrl ?? "https://moteurimmo.fr/api";
    62→    this.timeout = config.timeout ?? 30000;
    63→    this.maxRetries = config.maxRetries ?? 3;
    64→
    65→    // Rate limiter (300 requests/minute as per MoteurImmo limits)
    66→    this.rateLimiter = new RateLimiter(
    67→      config.rateLimitConfig ?? {
    68→        maxRequests: 300,
    69→        windowMs: 60000, // 1 minute
    70→      },
    71→    );
    72→
    73→    // Cache setup
    74→    this.cache = new MemoryCache(
    75→      config.cacheConfig?.defaultTtlMs ?? 300000, // 5 minutes default
    76→    );
    77→    this.cacheEnabled = config.cacheConfig?.enabled ?? true;
    78→  }

    81→  async searchListings(criteria: SearchCriteria): Promise<SearchResult> {
    82→    // Validate internal criteria
    83→    const validatedCriteria = SearchCriteriaSchema.parse(criteria);
    84→
    85→    // Convert to MoteurImmo API format
    86→    const apiCriteria =
    87→      this.convertToMoteurImmoSearchCriteria(validatedCriteria);
    88→
    89→    const cacheKey = this.cache.generateKey("search", apiCriteria);
    90→
    91→    try {
    92→      const result = await this.makeRequest<MoteurImmoSearchResponse>("/ads", {
    93→        method: "POST",
    94→        body: apiCriteria,
    95→        cacheKey,
    96→        cacheTtl: 300000, // 5 minute cache for search results
    97→      });
    98→
    99→      // Validate MoteurImmo response
   100→      const validatedResponse = MoteurImmoSearchResponseSchema.parse(result);
   101→
   102→      // Convert to internal format
   103→      return this.convertToInternalSearchResult(
   104→        validatedResponse,
   105→        validatedCriteria,
   106→      );
   107→    } catch (error) {
   108→      if (error instanceof Error && error.name === "ZodError") {
   109→        throw new MoteurImmoValidationError(
   110→          "Invalid search criteria or API response",
   111→          error,
   112→        );
   113→      }
   114→      throw error;
   115→    }

   291→  // Private methods
   292→  private async makeRequest<T>(
   293→    endpoint: string,
   294→    options: {
   295→      method?: "GET" | "POST" | "PUT" | "DELETE";
   296→      body?: unknown;
   297→      headers?: Record<string, string>;
   298→      skipRateLimit?: boolean;
   299→      cacheTtl?: number;
   300→      cacheKey?: string;
   301→    } = {},
   302→    queryParams?: Record<string, string>,
   303→  ): Promise<T> {
   304→    const {
   305→      method = "GET",
   306→      body,
   307→      headers = {},
   308→      skipRateLimit = false,
   309→      cacheTtl,
   310→      cacheKey,
   311→    } = options;
   312→
   313→    // Check cache first (for GET requests)
   314→    if (method === "GET" && this.cacheEnabled && cacheKey) {
   315→      const cachedResult = this.cache.get<T>(cacheKey);
   316→      if (cachedResult) {
   317→        logger.debug(`[MoteurImmo] Cache hit for ${cacheKey}`);
   318→        return cachedResult;
   319→      }
   320→    }
   321→
   322→    // Rate limiting check
   323→    if (!skipRateLimit) {
   324→      const rateLimitCheck = await this.rateLimiter.checkLimit();
   325→      if (!rateLimitCheck.allowed) {
   326→        throw new MoteurImmoRateLimitError(
   327→          `Rate limit exceeded. Retry after ${rateLimitCheck.retryAfter} seconds`,
   328→          rateLimitCheck.retryAfter,
   329→        );
   330→      }
   331→    }
   332→
   333→    // Build URL with query parameters for GET requests
   334→    let url = `${this.baseUrl}${endpoint}`;
   335→    if (method === "GET" || method === "DELETE") {
   336→      const searchParams = new URLSearchParams();
   337→      searchParams.append("apiKey", this.apiKey);
   338→
   339→      // Add additional query parameters if provided
   340→      if (queryParams) {
   341→        Object.entries(queryParams).forEach(([key, value]) => {
   342→          searchParams.append(key, value);
   343→        });
   344→      }
   345→
   346→      url += `?${searchParams.toString()}`;
   347→    }
   348→
   349→    const requestOptions: RequestInit = {
   350→      method,
   351→      headers: {
   352→        "Content-Type": "application/json",
   353→        "User-Agent": "ChooseWisely/1.0",
   354→        ...headers,
   355→      },
   356→      signal: AbortSignal.timeout(this.timeout),
   357→    };
   358→
   359→    // MoteurImmo uses apiKey in body for POST/PUT, and query param for GET/DELETE
   360→    if (body) {
   361→      // Add apiKey to body for POST/PUT requests
   362→      const bodyWithApiKey = {
   363→        apiKey: this.apiKey,
   364→        ...body,
   365→      };
   366→      requestOptions.body = JSON.stringify(bodyWithApiKey);
   367→    }
   368→
   369→    // Recursive function to handle retries
   370→    const executeRequest = async (attempt: number): Promise<T> => {
   371→      try {
   372→        logger.debug(`[MoteurImmo] ${method} ${url} (attempt ${attempt})`);
   373→
   374→        const response = await fetch(url, requestOptions);
   375→
   376→        // Handle different HTTP status codes
   377→        if (response.status === 429) {
   378→          const retryAfter = response.headers.get("Retry-After");
   379→          const retryAfterSeconds = retryAfter
   380→            ? parseInt(retryAfter)
   381→            : undefined;
   382→          throw new MoteurImmoRateLimitError(
   383→            "Rate limit exceeded by API",
   384→            retryAfterSeconds,
   385→          );
   386→        }
   387→
   388→        if (response.status === 401 || response.status === 403) {
   389→          throw new MoteurImmoAuthError("Authentication failed");
   390→        }
   391→
   392→        if (!response.ok) {
   393→          const errorData = await response.json().catch(() => ({}));
   394→          const apiError = APIErrorSchema.safeParse(errorData);
   395→
   396→          const errorMessage = apiError.success
   397→            ? apiError.data.message
   398→            : `HTTP ${response.status}: ${response.statusText}`;
   399→
   400→          throw new MoteurImmoError(
   401→            errorMessage,
   402→            apiError.success
   403→              ? (apiError.data.code ?? "API_ERROR")
   404→              : "HTTP_ERROR",
   405→            response.status,
   406→            errorData,
   407→          );
   408→        }
   409→
   410→        const data = await response.json();
   411→
   412→        // Cache successful GET responses
   413→        if (method === "GET" && this.cacheEnabled && cacheKey) {
   414→          this.cache.set(cacheKey, data, cacheTtl);
   415→          logger.debug(`[MoteurImmo] Cached response for ${cacheKey}`);
   416→        }
   417→
   418→        return data;
   419→      } catch (error) {
   420→        // Don't retry on certain errors
   421→        if (
   422→          error instanceof MoteurImmoAuthError ||
   423→          error instanceof MoteurImmoValidationError ||
   424→          (error instanceof MoteurImmoError && error.statusCode === 404)
   425→        ) {
   426→          throw error;
   427→        }
   428→
   429→        // If we've exhausted retries, throw the error
   430→        if (attempt >= this.maxRetries) {
   431→          // Convert fetch errors to our custom errors
   432→          if (error instanceof Error && error.name === "AbortError") {
   433→            throw new MoteurImmoNetworkError("Request timeout", error);
   434→          }
   435→
   436→          if (error instanceof TypeError && error.message.includes("fetch")) {
   437→            throw new MoteurImmoNetworkError("Network error", error);
   438→          }
   439→
   440→          throw error;
   441→        }
   442→
   443→        // For rate limit errors, wait before retrying
   444→        if (error instanceof MoteurImmoRateLimitError && error.retryAfter) {
   445→          logger.debug(
   446→            `[MoteurImmo] Rate limited, waiting ${error.retryAfter}s`,
   447→          );
   448→          const retryAfterMs = error.retryAfter * 1000;
   449→          return new Promise((resolve) => {
   450→            setTimeout(() => {
   451→              resolve(executeRequest(attempt + 1));
   452→            }, retryAfterMs);
   453→          });
   454→        }
   455→
   456→        // Exponential backoff for network errors
   457→        const delay = Math.min(1000 * Math.pow(2, attempt - 1), 10000);
   458→        logger.debug(`[MoteurImmo] Request failed, retrying in ${delay}ms`);
   459→        return new Promise((resolve) => {
   460→          setTimeout(() => {
   461→            resolve(executeRequest(attempt + 1));
   462→          }, delay);
   463→        });
   464→      }
   465→    };
   466→
   467→    return executeRequest(1);
   468→  }

   470→  // Data conversion methods
   471→  private convertToMoteurImmoSearchCriteria(
   472→    criteria: SearchCriteria,
   473→  ): MoteurImmoSearchCriteria {
   474→    const moteurImmoSearch: MoteurImmoSearchCriteria = {
   475→      apiKey: this.apiKey,
   476→      page: criteria.page,
   477→      maxLength: criteria.limit,
   478→      withCount: true, // Always get count for pagination
   479→      keywordsOperator: "and", // Default to 'and' operator
   480→    };
   481→
   482→    // Convert type
   483→    if (criteria.type) {
   484→      moteurImmoSearch.types = [criteria.type];
   485→    }
   486→
   487→    // Convert category
   488→    if (criteria.category) {
   489→      const categoryMap: Record<
   490→        string,
   491→        | "house"
   492→        | "flat"
   493→        | "land"
   494→        | "parking/garage/box"
   495→        | "office"
   496→        | "shop"
   497→        | "warehouse"
   498→        | "building"
   499→      > = {
   500→        house: "house",
   501→        flat: "flat",
   502→        land: "land",
   503→        parking: "parking/garage/box",
   504→      };
   505→      moteurImmoSearch.categories = [categoryMap[criteria.category]];
   506→    }
   507→
   508→    // Price filters
   509→    if (criteria.priceMin) moteurImmoSearch.priceMin = criteria.priceMin;
   510→    if (criteria.priceMax) moteurImmoSearch.priceMax = criteria.priceMax;
   511→
   512→    // Surface filters
   513→    if (criteria.surfaceMin) moteurImmoSearch.surfaceMin = criteria.surfaceMin;
   514→    if (criteria.surfaceMax) moteurImmoSearch.surfaceMax = criteria.surfaceMax;
   515→
   516→    // Rooms filters
   517→    if (criteria.roomsMin) moteurImmoSearch.roomsMin = criteria.roomsMin;
   518→    if (criteria.roomsMax) moteurImmoSearch.roomsMax = criteria.roomsMax;
   519→
   520→    // Location
   521→    if (criteria.city || criteria.postalCode) {
   522→      const location: { postalCode?: string } = {};
   523→
   524→      if (criteria.postalCode) {
   525→        location.postalCode = criteria.postalCode;
   526→      }
   527→
   528→      // If we have a city name, try to detect if it's a postal code or city name
   529→      if (criteria.city) {
   530→        const cityValue = criteria.city.trim();
   531→
   532→        // Check if it looks like a postal code (5 digits)
   533→        if (/^\d{5}$/.test(cityValue)) {
   534→          location.postalCode = cityValue;
   535→        } else {
   536→          // For city names, we'll use keywords instead of locations for better search
   537→          if (!criteria.search) {
   538→            moteurImmoSearch.keywords = [cityValue];
   539→            moteurImmoSearch.keywordsOperator = "and";
   540→          }
   541→        }
   542→      }
   543→
   544→      // Only add locations if we have a postal code
   545→      if (location.postalCode) {
   546→        moteurImmoSearch.locations = [location];
   547→
   548→        if (criteria.radius) {
   549→          moteurImmoSearch.radius = criteria.radius;
   550→        }
   551→      }
   552→    }
   553→
   554→    // Energy grades
   555→    if (criteria.energyGradeMin)
   556→      moteurImmoSearch.energyGradeMin = criteria.energyGradeMin;
   557→    if (criteria.energyGradeMax)
   558→      moteurImmoSearch.energyGradeMax = criteria.energyGradeMax;
   559→
   560→    // Options
   561→    if (criteria.options && criteria.options.length > 0) {
   562→      moteurImmoSearch.options = criteria.options;
   563→    }
   564→
   565→    // Keywords from search query
   566→    if (criteria.search) {
   567→      const searchKeywords = criteria.search.split(/\s+/).filter(Boolean);
   568→
   569→      // If we already have keywords from city processing, merge them
   570→      if (moteurImmoSearch.keywords) {
   571→        moteurImmoSearch.keywords = [
   572→          ...moteurImmoSearch.keywords,
   573→          ...searchKeywords,
   574→        ];
   575→      } else {
   576→        moteurImmoSearch.keywords = searchKeywords;
   577→      }
   578→      moteurImmoSearch.keywordsOperator = "and";
   579→    }
   580→
   581→    // Sorting
   582→    if (criteria.sortBy !== undefined) {
   583→      const sortMap: Record<
   584→        string,
   585→        | "creationDate-desc"
   586→        | "creationDate-asc"
   587→        | "price-asc"
   588→        | "price-desc"
   589→        | "pricePerSquareMeter-asc"
   590→        | "pricePerSquareMeter-desc"
   591→        | "lastPriceChangeDate-desc"
   592→        | "lastPriceChangeDate-asc"
   593→        | "lastEventDate-desc"
   594→        | "lastEventDate-asc"
   595→      > = {
   596→        price: criteria.sortOrder === "asc" ? "price-asc" : "price-desc",
   597→        date:
   598→          criteria.sortOrder === "asc"
   599→            ? "creationDate-asc"
   600→            : "creationDate-desc",
   601→        surface: criteria.sortOrder === "asc" ? "price-asc" : "price-desc", // Fallback to price as surface sort not available
   602→        pricePerM2:
   603→          criteria.sortOrder === "asc"
   604→            ? "pricePerSquareMeter-asc"
   605→            : "pricePerSquareMeter-desc",
   606→      };
   607→      moteurImmoSearch.sortBy = sortMap[criteria.sortBy] ?? "creationDate-desc";
   608→    }
   609→
   610→    return moteurImmoSearch;
   611→  }

   613→  private convertToInternalSearchResult(
   614→    response: MoteurImmoSearchResponse,
   615→    originalCriteria: SearchCriteria,
   616→  ): SearchResult {
   617→    const properties = response.ads.map((prop) =>
   618→      this.convertToInternalProperty(prop),
   619→    );
   620→
   621→    const page = originalCriteria.page;
   622→    const limit = originalCriteria.limit;
   623→    const total = response.count; // MoteurImmo returns total count in response
   624→    const totalPages = Math.ceil(total / limit);
   625→
   626→    return {
   627→      properties,
   628→      pagination: {
   629→        page,
   630→        limit,
   631→        total,
   632→        totalPages,
   633→        hasNext: page < totalPages,
   634→        hasPrev: page > 1,
   635→      },
   636→      filters: {
   637→        appliedFilters: originalCriteria,
   638→        availableTypes: ["sale", "rental"],
   639→        availableCategories: ["house", "flat", "land", "parking/garage/box"],
   640→        priceMin: 0,
   641→        priceMax: 10000000,
   642→        surfaceMin: 0,
   643→        surfaceMax: 1000,
   644→      },
   645→    };
   646→  }

   653→  private convertToInternalProperty(
   654→    moteurImmoProperty: MoteurImmoProperty,
   655→  ): Property {
   656→    return {
   657→      id: moteurImmoProperty.uniqueId,
   658→      uniqueId: moteurImmoProperty.uniqueId,
   659→      externalId: moteurImmoProperty.adId,
