    18→// Private helper to convert our PropertyAlert to MoteurImmo Alert format
    19→function convertToAlertCriteria(alert: CreateAlertRequest): AlertCriteria {
    20→  const searchCriteria = filtersToSearchCriteria({
    21→    ...alert.searchFilters,
    22→    page: 1,
    23→    sortBy: "",
    24→    sortOrder: "desc",
    25→    view: "list",
    26→  });
    27→
    28→  // Remove pagination fields as they're not allowed in alert criteria
    29→  const { page, limit, sortBy, sortOrder, ...alertCriteria } = searchCriteria;
    30→
    31→  return {
    32→    name: alert.name,
    33→    email: alert.email,
    34→    webhookUrl: alert.webhookUrl ?? process.env.MOTEURIMMO_WEBHOOK_URL,
    35→    watchedEvents: alert.watchedEvents,
    36→    ...alertCriteria,
    37→  } as AlertCriteria;
    38→}

    40→// Convert MoteurImmo Alert to our PropertyAlert format
    41→function convertToPropertyAlert(
    42→  moteurAlert: Alert,
    43→  userId: string,
    44→): PropertyAlert {
    45→  return {
    46→    id: moteurAlert.id,
    47→    name: moteurAlert.criteria.name,
    48→    userId: userId,
    49→    email: moteurAlert.criteria.email,
    50→    webhookUrl: moteurAlert.criteria.webhookUrl,
    51→    watchedEvents: moteurAlert.criteria
    52→      .watchedEvents as PropertyAlert["watchedEvents"],
    53→    searchFilters:
    54→      moteurAlert.criteria as unknown as PropertyAlert["searchFilters"],
    55→    isActive: moteurAlert.isActive,
    56→    createdAt: new Date(moteurAlert.createdAt),
    57→    updatedAt: new Date(moteurAlert.updatedAt),
    58→    lastTriggered: moteurAlert.lastTriggered
    59→      ? new Date(moteurAlert.lastTriggered)
    60→      : undefined,
    61→    triggerCount: moteurAlert.propertiesFound ?? 0,
    62→  };
    63→}

    65→export async function createAlertAction(
    66→  request: CreateAlertRequest,
    67→): Promise<PropertyAlert> {
    68→  const session = await auth.api.getSession({
    69→    headers: await headers(),
    70→  });
    71→
    72→  if (!session?.user?.id) {
    73→    redirect("/auth/signin");
    74→  }
    75→
    76→  try {
    77→    // Convert to MoteurImmo alert criteria format
    78→    const alertCriteria = convertToAlertCriteria(request);
    79→
    80→    logger.debug(
    81→      `🔔 [Alerts] Creating alert "${request.name}" for user ${session.user.id}`,
    82→      { criteria: alertCriteria },
    83→    );
    84→
    85→    // Register alert with MoteurImmo API
    86→    const moteurAlert = await moteurImmoClient.createAlert(alertCriteria);
    87→
    88→    // Convert back to our PropertyAlert format
    89→    const propertyAlert = convertToPropertyAlert(moteurAlert, session.user.id);
    90→
    91→    logger.debug(
    92→      `✅ [Alerts] Successfully created alert "${propertyAlert.name}" with ID ${propertyAlert.id}`,
    93→    );
    94→
    95→    return propertyAlert;
    96→  } catch (error) {
    97→    logger.error(
    98→      `❌ [Alerts] Failed to create alert "${request.name}" for user ${session.user.id}`,
    99→      { error },
   100→    );
   101→    throw new Error(
   102→      `Échec de la création de l'alerte : ${error instanceof Error ? error.message : "Erreur inconnue"}`,
   103→    );
   104→  }
   105→}

   107→export async function getAlertsAction(): Promise<PropertyAlert[]> {
   108→  const session = await auth.api.getSession({
   109→    headers: await headers(),
   110→  });
   111→
   112→  if (!session?.user?.id) {
   113→    redirect("/auth/signin");
   114→  }
   115→
   116→  try {
   117→    // Get all alerts from MoteurImmo API
   118→    const moteurAlerts = await moteurImmoClient.listAlerts();
   119→
   120→    // Convert to our format and filter by user
   121→    // Note: MoteurImmo API doesn't filter by user, so we need to do it client-side
   122→    // In production, you might want to store user-alert mappings in your database
   123→    const userAlerts = moteurAlerts
   124→      .map((alert) => convertToPropertyAlert(alert, session.user.id))
   125→      .filter((alert) => alert.userId === session.user.id);
   126→
   127→    logger.debug(
   128→      `🔔 [Alerts] Retrieved ${userAlerts.length} alerts for user ${session.user.id}`,
   129→    );
   130→
   131→    return userAlerts;
   132→  } catch (error) {
   133→    logger.error(
   134→      `❌ [Alerts] Failed to retrieve alerts for user ${session.user.id}`,
   135→      { error },
   136→    );
   137→    // Return empty array instead of throwing to avoid breaking the UI
   138→    return [];
   139→  }
   140→}

   142→export async function updateAlertAction(
   143→  request: UpdateAlertRequest,
   144→): Promise<PropertyAlert> {
   145→  const session = await auth.api.getSession({
   146→    headers: await headers(),
   147→  });
   148→
   149→  if (!session?.user?.id) {
   150→    redirect("/auth/signin");
   151→  }
   152→
   153→  try {
   154→    logger.debug(
   155→      `🔔 [Alerts] Updating alert ${request.id} for user ${session.user.id}`,
   156→      { updates: request },
   157→    );
   158→
   159→    // For MoteurImmo API, we need to delete and recreate the alert
   160→    // as it doesn't support direct updates
   161→    await moteurImmoClient.deleteAlert(request.id);
   162→
   163→    // Create the updated alert
   164→    const createRequest: CreateAlertRequest = {
   165→      name: request.name ?? "Updated Alert",
   166→      email: request.email,
   167→      webhookUrl: request.webhookUrl,
   168→      watchedEvents: request.watchedEvents ?? ["creation"],
   169→      searchFilters:
   170→        request.searchFilters ??
   171→        ({} as unknown as PropertyAlert["searchFilters"]),
   172→    };
   173→
   174→    const alertCriteria = convertToAlertCriteria(createRequest);
   175→    const moteurAlert = await moteurImmoClient.createAlert(alertCriteria);
   176→    const updatedAlert = convertToPropertyAlert(moteurAlert, session.user.id);
   177→
   178→    // Update isActive status if provided
   179→    if (typeof request.isActive !== "undefined" && !request.isActive) {
   180→      // If we need to deactivate, delete the alert
   181→      await moteurImmoClient.deleteAlert(moteurAlert.id);
   182→      updatedAlert.isActive = false;
   183→    }
   184→
   185→    logger.debug(
   186→      `✅ [Alerts] Successfully updated alert "${updatedAlert.name}"`,
   187→    );
   188→
   189→    return updatedAlert;
   190→  } catch (error) {
   191→    logger.error(
   192→      `❌ [Alerts] Failed to update alert ${request.id} for user ${session.user.id}`,
   193→      { error },
   194→    );
   195→    throw new Error(
   196→      `Échec de la mise à jour de l'alerte : ${error instanceof Error ? error.message : "Erreur inconnue"}`,
   197→    );
   198→  }
   199→}

   201→export async function deleteAlertAction(alertId: string): Promise<void> {
   202→  const session = await auth.api.getSession({
   203→    headers: await headers(),
   204→  });
   205→
   206→  if (!session?.user?.id) {
   207→    redirect("/auth/signin");
   208→  }
   209→
   210→  try {
   211→    logger.debug(
   212→      `🔔 [Alerts] Deleting alert ${alertId} for user ${session.user.id}`,
   213→    );
   214→
   215→    // Delete alert from MoteurImmo API
   216→    await moteurImmoClient.deleteAlert(alertId);
   217→
   218→    logger.debug(`✅ [Alerts] Successfully deleted alert ${alertId}`);
   219→  } catch (error) {
   220→    logger.error(
   221→      `❌ [Alerts] Failed to delete alert ${alertId} for user ${session.user.id}`,
   222→      { error },
   223→    );
   224→    throw new Error(
   225→      `Échec de la suppression de l'alerte : ${error instanceof Error ? error.message : "Erreur inconnue"}`,
   226→    );
   227→  }
   228→}
