    47→async function PropertyContent({
    48→  searchParams,
    49→}: {
    50→  searchParams: Record<string, string | string[] | undefined>;
    51→}) {
    52→  // Parse search params using nuqs server cache
    53→  const parsedFilters = searchParamsCache.parse(searchParams);
    54→
    55→  // Only search if we have some filters (avoid empty search on page load)
    56→  const hasActiveFilters =
    57→    parsedFilters.selectedCitiesJson ||
    58→    parsedFilters.selectedTypesJson ||
    59→    parsedFilters.selectedCategoriesJson ||
    60→    parsedFilters.priceMin ||
    61→    parsedFilters.priceMax ||
    62→    parsedFilters.surfaceMin ||
    63→    parsedFilters.surfaceMax ||
    64→    parsedFilters.rooms ||
    65→    parsedFilters.bedrooms ||
    66→    parsedFilters.city ||
    67→    parsedFilters.search;
    68→
    69→  let searchResult = { data: null, error: null } as {
    70→    data: SearchResult | null;
    71→    error: string | null;
    72→  };
    73→
    74→  if (hasActiveFilters) {
    75→    // Convert URL params to PropertySearchFilters only when needed
    76→    const filters: PropertySearchFilters = urlParamsToFilters(parsedFilters);
    77→    searchResult = await searchPropertiesAction(filters);
    78→  }
    79→
    80→  return (
    81→    <div className="container mx-auto px-4 py-6">
    82→      <div className="mb-8">
    83→        <h1 className="mb-2 text-3xl font-bold">
    84→          Recherche de biens immobiliers
    85→        </h1>
    86→        <p className="text-muted-foreground">
    87→          Trouvez le bien immobilier parfait grâce à notre moteur de recherche
    88→          avancé.
    89→        </p>
    90→      </div>
    91→
    92→      {/* Filters - Collapsible panel at top */}
    93→      <PropertyFiltersToggle searchResult={searchResult.data} />
    94→
    95→      {/* Results */}
    96→      <PropertyList
    97→        data={searchResult.data}
    98→        error={searchResult.error}
    99→        isLoading={false}
   100→      />
   101→    </div>
   102→  );
   103→}
