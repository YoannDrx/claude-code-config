   285→  /**
   286→   * Construit les paramètres de recherche
   287→   */
   288→  private buildSearchParams(request: DPESearchRequest): string {
   289→    const params = new URLSearchParams();
   290→    
   291→    if (request.address) {
   292→      params.append("q", request.address);
   293→    }
   294→    
   295→    if (request.postalCode) {
   296→      params.append("refine.code_postal", request.postalCode);
   297→    }
   298→    
   299→    if (request.city) {
   300→      params.append("refine.ville", request.city);
   301→    }
   302→    
   303→    if (request.coordinates) {
   304→      const [lng, lat] = request.coordinates;
   305→      params.append("geofilter.distance", `${lat},${lng},${request.radius}`);
   306→    }
   307→    
   308→    if (request.propertyType) {
   309→      params.append("refine.type_batiment", request.propertyType);
   310→    }
   311→    
   312→    if (request.minYearBuilt) {
   313→      params.append("refine.annee_construction_min", request.minYearBuilt.toString());
   314→    }
   315→    
   316→    if (request.maxYearBuilt) {
   317→      params.append("refine.annee_construction_max", request.maxYearBuilt.toString());
   318→    }
   319→    
   320→    if (request.minSurface) {
   321→      params.append("refine.surface_habitable_min", request.minSurface.toString());
   322→    }
   323→    
   324→    if (request.maxSurface) {
   325→      params.append("refine.surface_habitable_max", request.maxSurface.toString());
   326→    }
   327→    
   328→    params.append("rows", request.limit.toString());
   329→    params.append("start", "0");
   330→    
   331→    return params.toString();
   332→  }

   334→  /**
   335→   * Effectue une requête HTTP avec gestion des erreurs
   336→   */
   337→  private async makeRequest(endpoint: string, params?: string): Promise<DPERawResponse> {
   338→    const url = `${this.baseUrl}${endpoint}${params ? `?${params}` : ''}`;
   339→    
   340→    const headers: Record<string, string> = {
   341→      "Content-Type": "application/json",
   342→      "Accept": "application/json",
   343→    };
   344→    
   345→    if (this.apiKey) {
   346→      headers.Authorization = `Bearer ${this.apiKey}`;
   347→    }
   348→    
   349→    const controller = new AbortController();
   350→    const timeoutId = setTimeout(() => controller.abort(), this.timeout);
   351→    
   352→    try {
   353→      let response: Response;
   354→      let attempt = 0;
   355→
   356→      do {
   357→        attempt++;
   358→        // eslint-disable-next-line no-await-in-loop -- Retry logic requires sequential attempts
   359→        response = await fetch(url, {
   360→          method: "GET",
   361→          headers,
   362→          signal: controller.signal,
   363→        });
   364→
   365→        if (response.ok || attempt >= this.maxRetries) {
   366→          break;
   367→        }
   368→
   369→        // Attendre avant de réessayer (backoff exponentiel)
   370→        // eslint-disable-next-line no-await-in-loop -- Intentional delay between retry attempts
   371→        await new Promise(resolve =>
   372→          setTimeout(resolve, Math.pow(2, attempt) * 1000)
   373→        );
   374→      } while (attempt < this.maxRetries);
   375→      
   376→      if (!response.ok) {
   377→        const errorText = await response.text();
   378→        throw new Error(`HTTP ${response.status}: ${errorText}`);
   379→      }
   380→      
   381→      return await response.json();
   382→    } catch (error) {
   383→      if (error instanceof Error && error.name === "AbortError") {
   384→        throw new DPEAPIError("Timeout de la requête DPE");
   385→      }
   386→      throw error;
   387→    } finally {
   388→      clearTimeout(timeoutId);
   389→    }
   390→  }

   392→  /**
   393→   * Parse la réponse de recherche
   394→   */
   395→  private parseSearchResponse(rawResponse: DPERawResponse): DPESearchResponse {
   396→    const records = rawResponse.records ?? [];
   397→    const total = rawResponse.nhits ?? rawResponse.total_count ?? 0;
   398→
   399→    const results: DPE[] = [];
   400→
   401→    for (const record of records) {
   402→      try {
   403→        if (record.fields) {
   404→          const normalized = normalizeDPERawData({
   405→            ...record.fields,
   406→            id: record.recordid,
   407→          });
   408→          const validated = validateDPEData(normalized);
   409→          results.push(validated);
   410→        }
   411→      } catch (error) {
   412→        console.debug("Erreur de parsing d'un enregistrement DPE:", error);
   413→        // Continuer avec les autres enregistrements
   414→      }
   415→    }
   416→    
   417→    return {
   418→      results,
   419→      total,
   420→      page: 1,
   421→      pageSize: results.length,
   422→    };
   423→  }

   425→  /**
   426→   * Parse une réponse unique
   427→   */
   428→  private parseSingleResponse(record: { fields?: Record<string, unknown> } | Record<string, unknown>): DPE {
   429→    try {
   430→      const normalized = normalizeDPERawData((record as { fields?: Record<string, unknown> }).fields ?? record);
   431→      return validateDPEData(normalized);
   432→    } catch (error) {
   433→      throw new DPEAPIError(
   434→        `Erreur de validation des données DPE: ${error instanceof Error ? error.message : 'Erreur inconnue'}`
   435→      );
   436→    }
   437→  }

   439→  /**
   440→   * Gère le rate limiting
   441→   */
   442→  private async enforceRateLimit(): Promise<void> {
   443→    const now = Date.now();
   444→    const timeSinceLastRequest = now - this.lastRequestTime;
   445→    
   446→    if (timeSinceLastRequest < this.rateLimitDelay) {
   447→      await new Promise(resolve => 
   448→        setTimeout(resolve, this.rateLimitDelay - timeSinceLastRequest)
   449→      );
   450→    }
   451→
   452→    this.lastRequestTime = Date.now();
   453→  }
