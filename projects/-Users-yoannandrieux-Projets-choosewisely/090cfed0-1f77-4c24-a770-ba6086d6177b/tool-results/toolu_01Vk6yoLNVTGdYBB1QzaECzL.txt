     1→"use client";
     2→
     3→import { logger } from "@/lib/logger";
     4→import { useState } from "react";
     5→import { Button } from "@/components/ui/button";
     6→import { Badge } from "@/components/ui/badge";
     7→import { Textarea } from "@/components/ui/textarea";
     8→import {
     9→  Trash2,
    10→  Edit3,
    11→  MapPin,
    12→  Euro,
    13→  Home,
    14→  Save,
    15→  X,
    16→  ExternalLink,
    17→  Star,
    18→  ChevronDown,
    19→  ChevronUp,
    20→} from "lucide-react";
    21→import { DPECard } from "@/components/energy/dpe-card";
    22→import { useDPE } from "@/hooks/use-dpe";
    23→import { deleteFavoriteAction, updateFavoriteAction } from "./favorites.action";
    24→import { cn } from "@/lib/utils";
    25→import type { Favorite, Property } from "@/generated/prisma";
    26→
    27→type FavoriteWithProperty = Favorite & {
    28→  property: Property;
    29→};
    30→
    31→type FavoriteCardProps = {
    32→  favorite: FavoriteWithProperty;
    33→  onUpdate: () => void;
    34→};
    35→
    36→export function FavoriteCard({ favorite, onUpdate }: FavoriteCardProps) {
    37→  const [isLoading, setIsLoading] = useState(false);
    38→  const [isEditing, setIsEditing] = useState(false);
    39→  const [notes, setNotes] = useState(favorite.notes ?? "");
    40→  const [priority, setPriority] = useState<"high" | "medium" | "low">(
    41→    favorite.priority as "high" | "medium" | "low",
    42→  );
    43→
    44→  const { isLoading: dpeLoading, enrichFavorite, formatDPEData } = useDPE();
    45→  const [dpeExpanded, setDpeExpanded] = useState(false);
    46→
    47→  const dpeData = formatDPEData(favorite);
    48→
    49→  const handleRefreshDPE = async () => {
    50→    try {
    51→      await enrichFavorite(favorite.id);
    52→      onUpdate();
    53→    } catch (error) {
    54→      logger.debug("Failed to refresh DPE data:", error);
    55→    }
    56→  };
    57→
    58→  const handleDelete = async () => {
    59→    if (
    60→      !confirm(
    61→        `Êtes-vous sûr de vouloir supprimer "${favorite.property.title}" de vos favoris ?`,
    62→      )
    63→    ) {
    64→      return;
    65→    }
    66→
    67→    setIsLoading(true);
    68→    try {
    69→      await deleteFavoriteAction(favorite.id);
    70→      onUpdate();
    71→    } catch (error) {
    72→      logger.debug("Erreur lors de la suppression:", error);
    73→    } finally {
    74→      setIsLoading(false);
    75→    }
    76→  };
    77→
    78→  const handleSave = async () => {
    79→    setIsLoading(true);
    80→    try {
    81→      await updateFavoriteAction({
    82→        id: favorite.id,
    83→        notes: notes,
    84→        priority: priority,
    85→      });
    86→      setIsEditing(false);
    87→      onUpdate();
    88→    } catch (error) {
    89→      logger.debug("Erreur lors de la sauvegarde:", error);
    90→    } finally {
    91→      setIsLoading(false);
    92→    }
    93→  };
    94→
    95→  const handleCancel = () => {
    96→    setIsEditing(false);
    97→    setNotes(favorite.notes ?? "");
    98→    setPriority(favorite.priority as "high" | "medium" | "low");
    99→  };
   100→
   101→  const handleStatusChange = async (newStatus: string) => {
   102→    setIsLoading(true);
   103→    try {
   104→      await updateFavoriteAction({
   105→        id: favorite.id,
   106→        status: newStatus as "saved" | "visited" | "contacted" | "rejected",
   107→      });
   108→      onUpdate();
   109→    } catch (error) {
   110→      logger.debug("Erreur lors de la mise à jour:", error);
   111→    } finally {
   112→      setIsLoading(false);
   113→    }
   114→  };
   115→
   116→  const getPriorityVariant = (priority: string) => {
   117→    switch (priority) {
   118→      case "high":
   119→        return "destructive";
   120→      case "low":
   121→        return "secondary";
   122→      default:
   123→        return "accent";
   124→    }
   125→  };
   126→
   127→  const getStatusLabel = (status: string) => {
   128→    switch (status) {
   129→      case "saved":
   130→        return "Sauvegardé";
   131→      case "visited":
   132→        return "Visité";
   133→      case "contacted":
   134→        return "Contacté";
   135→      case "rejected":
   136→        return "Rejeté";
   137→      default:
   138→        return status;
   139→    }
   140→  };
   141→
   142→  return (
   143→    <div className="glass-card overflow-hidden transition-all duration-300 hover:shadow-[0_10px_30px_rgba(0,0,0,0.3)]">
   144→      <div className="p-5 space-y-4">
   145→        <div className="flex items-start justify-between gap-4">
   146→          <div className="min-w-0 flex-1 space-y-3">
   147→            <div className="flex items-center gap-3">
   148→              <h3 className="truncate text-lg font-semibold text-foreground">
   149→                {favorite.property.title}
   150→              </h3>
   151→              <Badge
   152→                variant={getPriorityVariant(favorite.priority)}
   153→                className="flex-shrink-0"
   154→              >
   155→                {favorite.priority === "high" && (
   156→                  <Star className="mr-1 size-3" />
   157→                )}
   158→                {favorite.priority === "high"
   159→                  ? "Priorité haute"
   160→                  : favorite.priority === "low"
   161→                    ? "Priorité basse"
   162→                    : "Normale"}
   163→              </Badge>
   164→            </div>
   165→
   166→            <div className="space-y-2">
   167→              <div className="flex items-center gap-2 text-sm text-muted-foreground">
   168→                <MapPin className="size-4 text-primary" />
   169→                {favorite.property.address}, {favorite.property.city} (
   170→                {favorite.property.postalCode})
   171→              </div>
   172→
   173→              <div className="flex items-center gap-4 text-sm">
   174→                <div className="flex items-center gap-1.5 text-primary font-semibold">
   175→                  <Euro className="size-4" />
   176→                  {favorite.property.price.toLocaleString("fr-FR")}€
   177→                </div>
   178→                {favorite.property.surface && (
   179→                  <div className="flex items-center gap-1.5 text-muted-foreground">
   180→                    <Home className="size-4" />
   181→                    {favorite.property.surface}m²
   182→                  </div>
   183→                )}
   184→                {favorite.property.rooms && (
   185→                  <div className="text-muted-foreground">
   186→                    {favorite.property.rooms} pièces
   187→                  </div>
   188→                )}
   189→              </div>
   190→            </div>
   191→
   192→            <div className="flex flex-wrap items-center gap-2">
   193→              {["saved", "visited", "contacted", "rejected"].map((status) => (
   194→                <button
   195→                  key={status}
   196→                  onClick={async () => handleStatusChange(status)}
   197→                  disabled={isLoading}
   198→                  className={cn(
   199→                    "inline-flex items-center rounded-lg px-3 py-1.5 text-xs font-medium transition-all duration-200",
   200→                    favorite.status === status
   201→                      ? "bg-primary text-primary-foreground shadow-[0_0_15px_var(--glow-accent)]"
   202→                      : "bg-white/5 text-muted-foreground hover:bg-white/10 hover:text-foreground border border-[var(--border-subtle)]",
   203→                  )}
   204→                >
   205→                  {getStatusLabel(status)}
   206→                </button>
   207→              ))}
   208→            </div>
   209→          </div>
   210→        </div>
   211→
   212→        <div className="space-y-3">
   213→          <button
   214→            onClick={() => setDpeExpanded(!dpeExpanded)}
   215→            className="flex items-center justify-between w-full px-4 py-3 rounded-xl bg-white/5 hover:bg-white/10 transition-colors duration-200"
   216→          >
   217→            <div className="flex items-center gap-3">
   218→              <span className="text-sm font-medium text-foreground">
   219→                Performance Énergétique
   220→              </span>
   221→              {dpeData.dpeClass && (
   222→                <div
   223→                  className={cn(
   224→                    "size-7 rounded-lg flex items-center justify-center text-xs font-bold text-white",
   225→                    dpeData.dpeClass === "A" || dpeData.dpeClass === "B"
   226→                      ? "bg-emerald-500"
   227→                      : dpeData.dpeClass === "C" || dpeData.dpeClass === "D"
   228→                        ? "bg-yellow-500"
   229→                        : "bg-red-500",
   230→                  )}
   231→                >
   232→                  {dpeData.dpeClass}
   233→                </div>
   234→              )}
   235→              {dpeData.rentalRestrictionRisk && (
   236→                <Badge
   237→                  variant={
   238→                    dpeData.rentalRestrictionRisk === "critical" ||
   239→                    dpeData.rentalRestrictionRisk === "high"
   240→                      ? "destructive"
   241→                      : dpeData.rentalRestrictionRisk === "medium"
   242→                        ? "warning"
   243→                        : "secondary"
   244→                  }
   245→                  className="text-xs"
   246→                >
   247→                  Risque:{" "}
   248→                  {dpeData.rentalRestrictionRisk === "critical"
   249→                    ? "Critique"
   250→                    : dpeData.rentalRestrictionRisk === "high"
   251→                      ? "Élevé"
   252→                      : dpeData.rentalRestrictionRisk === "medium"
   253→                        ? "Moyen"
   254→                        : "Faible"}
   255→                </Badge>
   256→              )}
   257→            </div>
   258→            {dpeExpanded ? (
   259→              <ChevronUp className="size-4 text-muted-foreground" />
   260→            ) : (
   261→              <ChevronDown className="size-4 text-muted-foreground" />
   262→            )}
   263→          </button>
   264→
   265→          {dpeExpanded && (
   266→            <div className="animate-fade-in">
   267→              <DPECard
   268→                data={dpeData}
   269→                isLoading={dpeLoading}
   270→                onRefresh={handleRefreshDPE}
   271→                compact
   272→              />
   273→            </div>
   274→          )}
   275→        </div>
   276→
   277→        <div className="space-y-2">
   278→          <h4 className="text-sm font-medium text-foreground">
   279→            Notes personnelles
   280→          </h4>
   281→          {isEditing ? (
   282→            <div className="space-y-3">
   283→              <Textarea
   284→                value={notes}
   285→                onChange={(e) => setNotes(e.target.value)}
   286→                placeholder="Ajoutez vos notes sur ce bien..."
   287→                className="min-h-[80px] bg-white/5 border-[var(--border-subtle)]"
   288→                maxLength={500}
   289→              />
   290→              <div className="flex items-center justify-between">
   291→                <p className="text-xs text-muted-foreground">
   292→                  {notes.length}/500 caractères
   293→                </p>
   294→                <div className="flex items-center gap-2">
   295→                  <Button
   296→                    onClick={handleCancel}
   297→                    variant="outline"
   298→                    size="sm"
   299→                    disabled={isLoading}
   300→                  >
   301→                    <X className="mr-1 size-3" />
   302→                    Annuler
   303→                  </Button>
   304→                  <Button
   305→                    onClick={handleSave}
   306→                    size="sm"
   307→                    variant="accent"
   308→                    disabled={isLoading}
   309→                  >
   310→                    <Save className="mr-1 size-3" />
   311→                    Sauvegarder
   312→                  </Button>
   313→                </div>
   314→              </div>
   315→            </div>
   316→          ) : (
   317→            <div>
   318→              {favorite.notes ? (
   319→                <p className="rounded-xl bg-white/5 p-4 text-sm text-muted-foreground whitespace-pre-wrap">
   320→                  {favorite.notes}
   321→                </p>
   322→              ) : (
   323→                <p className="text-sm text-muted-foreground italic">
   324→                  Aucune note pour ce bien
   325→                </p>
   326→              )}
   327→            </div>
   328→          )}
   329→        </div>
   330→
   331→        <div className="flex items-center justify-between pt-3 border-t border-[var(--border-subtle)]">
   332→          <div className="text-xs text-muted-foreground">
   333→            Ajouté le{" "}
   334→            {new Date(favorite.createdAt).toLocaleDateString("fr-FR")}
   335→          </div>
   336→
   337→          <div className="flex items-center gap-1">
   338→            {favorite.property.sourceUrl && (
   339→              <Button
   340→                asChild
   341→                variant="ghost"
   342→                size="icon"
   343→                className="size-8"
   344→                title="Voir l'annonce"
   345→              >
   346→                <a
   347→                  href={favorite.property.sourceUrl}
   348→                  target="_blank"
   349→                  rel="noopener noreferrer"
   350→                >
   351→                  <ExternalLink className="size-4" />
   352→                </a>
   353→              </Button>
   354→            )}
   355→
   356→            <Button
   357→              onClick={() => setIsEditing(!isEditing)}
   358→              variant="ghost"
   359→              size="icon"
   360→              className="size-8"
   361→              disabled={isLoading}
   362→              title="Modifier les notes"
   363→            >
   364→              <Edit3 className="size-4" />
   365→            </Button>
   366→
   367→            <Button
   368→              onClick={handleDelete}
   369→              variant="ghost"
   370→              size="icon"
   371→              className="size-8 text-destructive hover:text-destructive hover:bg-destructive/10"
   372→              disabled={isLoading}
   373→              title="Supprimer des favoris"
   374→            >
   375→              <Trash2 className="size-4" />
   376→            </Button>
   377→          </div>
   378→        </div>
   379→      </div>
   380→    </div>
   381→  );
   382→}
   383→
