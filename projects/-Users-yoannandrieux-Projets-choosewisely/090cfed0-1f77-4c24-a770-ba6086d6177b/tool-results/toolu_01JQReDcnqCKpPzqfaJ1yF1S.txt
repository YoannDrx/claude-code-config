    11→function useBoundedScroll(threshold: number) {
    12→  const { scrollY } = useScroll();
    13→  const scrollYBounded = useMotionValue(0);
    14→  const scrollYBoundedProgress = useTransform(
    15→    scrollYBounded,
    16→    [0, threshold],
    17→    [0, 1],
    18→  );
    19→
    20→  useEffect(() => {
    21→    const onChange = (current: number) => {
    22→      const previous = scrollY.getPrevious() ?? 0;
    23→      const diff = current - previous;
    24→      const newScrollYBounded = scrollYBounded.get() + diff;
    25→
    26→      scrollYBounded.set(clamp(newScrollYBounded, 0, threshold));
    27→    };
    28→
    29→    const deleteEvent = scrollY.on("change", onChange);
    30→
    31→    const listener = () => {
    32→      const currentScroll = window.scrollY;
    33→      onChange(currentScroll);
    34→    };
    35→
    36→    window.addEventListener("scroll", listener);
    37→
    38→    return () => {
    39→      deleteEvent();
    40→      window.removeEventListener("scroll", listener);
    41→    };
    42→  }, [threshold, scrollY, scrollYBounded]);
    43→
    44→  return { scrollYBounded, scrollYBoundedProgress };
    45→}
