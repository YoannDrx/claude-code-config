     1→import * as React from "react";
     2→import { cn } from "@/lib/utils";
     3→import { TrendingUp, TrendingDown, Minus } from "lucide-react";
     4→
     5→type StatsCardProps = {
     6→  label: string;
     7→  value: string | number;
     8→  trend?: {
     9→    value: number;
    10→    label?: string;
    11→  };
    12→  icon?: React.ReactNode;
    13→  variant?: "default" | "accent" | "highlight";
    14→  className?: string;
    15→};
    16→
    17→function StatsCard({
    18→  label,
    19→  value,
    20→  trend,
    21→  icon,
    22→  variant = "default",
    23→  className,
    24→}: StatsCardProps) {
    25→  const trendDirection =
    26→    trend?.value === 0 ? "neutral" : trend?.value && trend.value > 0 ? "up" : "down";
    27→
    28→  const variantStyles = {
    29→    default: "glass-card",
    30→    accent:
    31→      "glass-card border-[var(--brand-accent)]/20 shadow-[0_0_30px_var(--glow-accent)]",
    32→    highlight:
    33→      "glass-card border-[var(--brand-highlight)]/20 shadow-[0_0_30px_var(--glow-highlight)]",
    34→  };
    35→
    36→  return (
    37→    <div
    38→      className={cn(
    39→        "relative overflow-hidden p-6 transition-all duration-300 hover:scale-[1.02]",
    40→        variantStyles[variant],
    41→        className,
    42→      )}
    43→    >
    44→      <div className="flex items-start justify-between">
    45→        <div className="flex-1 space-y-3">
    46→          <p className="text-sm font-medium text-muted-foreground">{label}</p>
    47→          <p className="text-3xl font-bold tracking-tight text-foreground">
    48→            {value}
    49→          </p>
    50→          {trend && (
    51→            <div className="flex items-center gap-1.5">
    52→              {trendDirection === "up" && (
    53→                <TrendingUp className="size-4 text-success" />
    54→              )}
    55→              {trendDirection === "down" && (
    56→                <TrendingDown className="size-4 text-destructive" />
    57→              )}
    58→              {trendDirection === "neutral" && (
    59→                <Minus className="size-4 text-muted-foreground" />
    60→              )}
    61→              <span
    62→                className={cn(
    63→                  "text-sm font-medium",
    64→                  trendDirection === "up" && "text-success",
    65→                  trendDirection === "down" && "text-destructive",
    66→                  trendDirection === "neutral" && "text-muted-foreground",
    67→                )}
    68→              >
    69→                {trend.value > 0 ? "+" : ""}
    70→                {trend.value}%
    71→              </span>
    72→              {trend.label && (
    73→                <span className="text-sm text-muted-foreground">
    74→                  {trend.label}
    75→                </span>
    76→              )}
    77→            </div>
    78→          )}
    79→        </div>
    80→        {icon && (
    81→          <div
    82→            className={cn(
    83→              "flex size-12 items-center justify-center rounded-xl",
    84→              variant === "accent" &&
    85→                "bg-[var(--brand-accent)]/10 text-[var(--brand-accent)]",
    86→              variant === "highlight" &&
    87→                "bg-[var(--brand-highlight)]/10 text-[var(--brand-highlight)]",
    88→              variant === "default" && "bg-muted text-muted-foreground",
    89→            )}
    90→          >
    91→            {icon}
    92→          </div>
    93→        )}
    94→      </div>
    95→    </div>
    96→  );
    97→}
    98→
    99→type MiniStatsProps = {
   100→  label: string;
   101→  value: string | number;
   102→  icon?: React.ReactNode;
   103→  className?: string;

   105→
   106→function MiniStats({ label, value, icon, className }: MiniStatsProps) {
   107→  return (

   110→        "flex items-center gap-3 rounded-xl bg-white/5 px-4 py-3",

   115→        <div className="flex size-8 items-center justify-center rounded-lg bg-primary/10 text-primary">
   116→          {icon}

   118→      )}
   119→      <div className="flex-1">
