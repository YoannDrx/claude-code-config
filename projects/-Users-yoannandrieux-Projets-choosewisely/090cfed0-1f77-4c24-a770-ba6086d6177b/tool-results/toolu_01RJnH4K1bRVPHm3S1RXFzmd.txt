    23→function DPEScale({
    24→  grade,
    25→  showLabel = true,
    26→  size = "md",
    27→  className,
    28→}: DPEScaleProps) {
    29→  const normalizedGrade = grade?.toUpperCase() as DPEGrade;
    30→  const activeIndex = DPE_GRADES.findIndex((g) => g.grade === normalizedGrade);
    31→
    32→  const sizeClasses = {
    33→    sm: "h-4 text-xs",
    34→    md: "h-6 text-sm",
    35→    lg: "h-8 text-base",
    36→  };
    37→
    38→  const widthClasses = {
    39→    sm: ["w-6", "w-7", "w-8", "w-9", "w-10", "w-11", "w-12"],

    41→    lg: ["w-10", "w-12", "w-14", "w-16", "w-18", "w-20", "w-22"],
    42→  };
    43→
    44→  return (
    45→    <div className={cn("flex flex-col gap-0.5", className)}>
    46→      {DPE_GRADES.map((dpe, index) => {
    47→        const isActive = index === activeIndex;
    48→        return (
    49→          <div
    50→            key={dpe.grade}
    51→            className={cn(
    52→              "flex items-center transition-all duration-300",
    53→              sizeClasses[size],

    55→              isActive ? "scale-105" : "opacity-60",
    56→            )}
    57→          >
    58→            <div
    59→              className={cn(
    60→                "flex h-full w-full items-center justify-between rounded-r-md px-2 font-semibold text-white",
    61→                isActive && "shadow-lg",
    62→              )}
    63→              style={{
    64→                backgroundColor: dpe.color,
    65→                boxShadow: isActive ? `0 0 20px ${dpe.color}50` : undefined,
    66→              }}
    67→            >
    68→              <span>{dpe.label}</span>
    69→              {isActive && showLabel && (
    70→                <span className="ml-2 text-white/90">
    71→                  {normalizedGrade}
    72→                </span>
    73→              )}
    74→            </div>
    75→          </div>
    76→        );
    77→      })}
    78→    </div>
    79→  );
