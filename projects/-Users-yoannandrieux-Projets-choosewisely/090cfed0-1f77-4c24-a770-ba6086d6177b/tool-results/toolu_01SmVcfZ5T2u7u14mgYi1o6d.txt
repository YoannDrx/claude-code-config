     1→"use client";
     2→
     3→import { useState, useEffect, useTransition } from "react";
     4→import { Button } from "@/components/ui/button";
     5→import { Card, CardContent, CardHeader, CardTitle } from "@/components/ui/card";
     6→import { Badge } from "@/components/ui/badge";
     7→import type { Property } from "@/lib/moteurimmo/types";
     8→import { PropertyImageGallery } from "./property-image-gallery";
     9→import { PropertyInfoPanel } from "./property-info-panel";
    10→import { PropertyMapboxLocation } from "./property-mapbox-location";
    11→import { DVFAnalysisComponent } from "./components/dvf-analysis";
    12→import { NegotiationTrigger } from "@/features/negotiation/components/NegotiationTrigger";
    13→import { useComparisonStore } from "@/features/comparison/comparison-store";
    14→import {
    15→  toggleFavoriteAction,
    16→  getFavoritesAction,
    17→} from "@/features/favorites/favorites.action";
    18→import { analyzeDVFAction } from "./property-dvf.action";
    19→import type { DVFAnalysis } from "@/lib/services/dvf/types";
    20→import {
    21→  ArrowLeft,
    22→  Heart,
    23→  ExternalLink,
    24→  Share,
    25→  Scale,
    26→  Calculator,
    27→  Loader2,
    28→  Check,
    29→  Zap,
    30→} from "lucide-react";
    31→import Link from "next/link";
    32→import { toast } from "sonner";
    33→
    34→type PropertyDetailViewProps = {
    35→  property: Property;
    36→};
    37→
    38→export function PropertyDetailView({ property }: PropertyDetailViewProps) {
    39→  const [isFavorite, setIsFavorite] = useState(false);
    40→  const [favoriteLoading, setFavoriteLoading] = useState(false);
    41→  const [dvfAnalysis, setDvfAnalysis] = useState<DVFAnalysis | null>(null);
    42→  const [dvfLoading, setDvfLoading] = useState(true);
    43→  const [dvfError, setDvfError] = useState<string | null>(null);
    44→  const [isPending, startTransition] = useTransition();
    45→
    46→  const { addProperty, removeProperty, hasProperty } = useComparisonStore();
    47→  const isInComparison = hasProperty(property.uniqueId);
    48→
    49→  // Check if property is in favorites on mount
    50→  useEffect(() => {
    51→    const checkFavoriteStatus = async () => {
    52→      try {
    53→        const favorites = await getFavoritesAction();
    54→        const isFav = favorites.some(
    55→          (f) => f.property.externalId === property.uniqueId
    56→        );
    57→        setIsFavorite(isFav);
    58→      } catch {
    59→        // User might not be logged in
    60→      }
    61→    };
    62→    void checkFavoriteStatus();
    63→  }, [property.uniqueId]);
    64→
    65→  // Load DVF analysis
    66→  useEffect(() => {
    67→    const loadDVFAnalysis = async () => {
    68→      setDvfLoading(true);
    69→      setDvfError(null);
    70→
    71→      const result = await analyzeDVFAction(property);
    72→
    73→      if (result.error) {
    74→        setDvfError(result.error);
    75→      } else {
    76→        setDvfAnalysis(result.analysis);
    77→      }
    78→
    79→      setDvfLoading(false);
    80→    };
    81→
    82→    void loadDVFAnalysis();
    83→  }, [property]);
    84→
    85→  const handleShare = async () => {
    86→    if ("share" in navigator) {
    87→      try {
    88→        await navigator.share({
    89→          title: property.title,
    90→          text: property.description,
    91→          url: window.location.href,
    92→        });
    93→        toast.success("Lien partagé");
    94→        return;
    95→      } catch {
    96→        // Fall through to clipboard fallback
    97→      }
    98→    }
    99→
   100→    // Fallback to clipboard
   101→    await navigator.clipboard.writeText(window.location.href);
   102→    toast.success("Lien copié dans le presse-papier");
   103→  };
   104→
   105→  const handleToggleFavorite = async () => {
   106→    setFavoriteLoading(true);
   107→    try {
   108→      const result = await toggleFavoriteAction(property.uniqueId);
   109→      setIsFavorite(result.isFavorite);
   110→      toast.success(
   111→        result.isFavorite ? "Ajouté aux favoris" : "Retiré des favoris"
   112→      );
   113→    } catch (error) {
   114→      toast.error(
   115→        error instanceof Error ? error.message : "Erreur lors de l'opération"
   116→      );
   117→    } finally {
   118→      setFavoriteLoading(false);
   119→    }
   120→  };
   121→
   122→  const handleToggleComparison = () => {
   123→    startTransition(() => {
   124→      if (isInComparison) {
   125→        removeProperty(property.uniqueId);
   126→        toast.success("Retiré de la comparaison");
   127→      } else {
   128→        const comparisonProperty = {
   129→          ...property,
   130→        };
   131→        addProperty(comparisonProperty);
   132→        toast.success("Ajouté à la comparaison");
   133→      }
   134→    });
   135→  };
   136→
   137→  const formatPrice = (price: number) => {
   138→    return new Intl.NumberFormat("fr-FR", {
   139→      style: "currency",
   140→      currency: "EUR",
   141→      maximumFractionDigits: 0,
   142→    }).format(price);
   143→  };
   144→
   145→  return (
   146→    <div className="container mx-auto px-4 py-6">
   147→      {/* Header with navigation */}
   148→      <div className="mb-6 flex items-center justify-between">
   149→        <Button
   150→          variant="ghost"
   151→          size="sm"
   152→          asChild
   153→          className="flex items-center gap-2"
   154→        >
   155→          <Link href="/properties">
   156→            <ArrowLeft className="h-4 w-4" />
   157→            Retour à la recherche
   158→          </Link>
   159→        </Button>
   160→
   161→        {/* Quick actions in header */}
   162→        <div className="flex items-center gap-2">
   163→          <Button
   164→            variant={isFavorite ? "default" : "outline"}
   165→            size="sm"
   166→            onClick={handleToggleFavorite}
   167→            disabled={favoriteLoading}
   168→          >
   169→            {favoriteLoading ? (
   170→              <Loader2 className="h-4 w-4 animate-spin" />
   171→            ) : isFavorite ? (
   172→              <Check className="h-4 w-4" />
   173→            ) : (
   174→              <Heart className="h-4 w-4" />
   175→            )}
   176→          </Button>
   177→          <Button
   178→            variant={isInComparison ? "default" : "outline"}
   179→            size="sm"
   180→            onClick={handleToggleComparison}
   181→            disabled={isPending}
   182→          >
   183→            {isPending ? (
   184→              <Loader2 className="h-4 w-4 animate-spin" />
   185→            ) : (
   186→              <Scale className="h-4 w-4" />
   187→            )}
   188→          </Button>
   189→        </div>
   190→      </div>
   191→
   192→      {/* Property title and price */}
   193→      <div className="mb-6">
   194→        <div className="flex flex-wrap items-start justify-between gap-4">
   195→          <div>
   196→            <h1 className="mb-2 text-2xl font-bold">{property.title}</h1>
   197→            <p className="text-muted-foreground">
   198→              {property.city}
   199→              {property.postalCode && ` (${property.postalCode})`}
   200→            </p>
   201→          </div>
   202→          <div className="text-right">
   203→            <div className="text-3xl font-bold text-primary">
   204→              {formatPrice(property.price)}
   205→            </div>
   206→            {property.pricePerM2 && (
   207→              <div className="text-muted-foreground text-sm">
   208→                {formatPrice(property.pricePerM2)}/m²
   209→              </div>
   210→            )}
   211→            {property.priceDrop && property.priceDrop > 0 && (
   212→              <Badge variant="destructive" className="mt-1">
   213→                -{property.priceDrop}%
   214→              </Badge>
   215→            )}
   216→          </div>
   217→        </div>
   218→      </div>
   219→
   220→      {/* Main content grid */}
   221→      <div className="grid gap-6 lg:grid-cols-3">
   222→        {/* Left column - Images and description */}
   223→        <div className="space-y-6 lg:col-span-2">
   224→          {/* Image gallery */}
   225→          <PropertyImageGallery images={property.images} title={property.title} />
   226→
   227→          {/* Description */}
   228→          {property.description && (
   229→            <Card>
   230→              <CardHeader>
   231→                <CardTitle>Description</CardTitle>
   232→              </CardHeader>
   233→              <CardContent>
   234→                <div className="prose prose-sm max-w-none">
   235→                  <p className="whitespace-pre-line">{property.description}</p>
   236→                </div>
   237→              </CardContent>
   238→            </Card>
   239→          )}
   240→
   241→          {/* DVF Analysis */}
   242→          <DVFAnalysisComponent
   243→            analysis={dvfAnalysis}
   244→            loading={dvfLoading}
   245→            error={dvfError}
   246→            propertyAddress={`${property.address}, ${property.city}`}
   247→          />
   248→
   249→          {/* Location map */}
   250→          <PropertyMapboxLocation property={property} />
   251→        </div>
   252→
   253→        {/* Right column - Property info and actions */}
   254→        <div className="space-y-6">
   255→          <PropertyInfoPanel property={property} />
   256→
   257→          {/* Main action buttons */}
   258→          <div className="space-y-3">
   259→            <Button
   260→              className="w-full"
   261→              size="lg"
   262→              variant={isFavorite ? "secondary" : "default"}
   263→              onClick={handleToggleFavorite}
   264→              disabled={favoriteLoading}
   265→            >
   266→              {favoriteLoading ? (
   267→                <Loader2 className="mr-2 h-4 w-4 animate-spin" />
   268→              ) : isFavorite ? (
   269→                <Check className="mr-2 h-4 w-4" />
   270→              ) : (
   271→                <Heart className="mr-2 h-4 w-4" />
   272→              )}
   273→              {isFavorite ? "Dans vos favoris" : "Sauvegarder en favoris"}
   274→            </Button>
   275→
   276→            <Button
   277→              className="w-full"
   278→              size="lg"
   279→              variant={isInComparison ? "secondary" : "outline"}
   280→              onClick={handleToggleComparison}
   281→              disabled={isPending}
   282→            >
   283→              {isPending ? (
   284→                <Loader2 className="mr-2 h-4 w-4 animate-spin" />
   285→              ) : (
   286→                <Scale className="mr-2 h-4 w-4" />
   287→              )}
   288→              {isInComparison ? "Dans la comparaison" : "Ajouter à la comparaison"}
   289→            </Button>
   290→
   291→            <div className="grid grid-cols-2 gap-2">
   292→              <Button
   293→                variant="outline"
   294→                size="sm"
   295→                onClick={handleShare}
   296→                className="flex items-center gap-2"
   297→              >
   298→                <Share className="h-4 w-4" />
   299→                Partager
   300→              </Button>
   301→
   302→              {property.url && (
   303→                <Button
   304→                  variant="outline"
   305→                  size="sm"
   306→                  asChild
   307→                  className="flex items-center gap-2"
   308→                >
   309→                  <Link href={property.url} target="_blank">
   310→                    <ExternalLink className="h-4 w-4" />
   311→                    Source
   312→                  </Link>
   313→                </Button>
   314→              )}
   315→            </div>
   316→          </div>
   317→
   318→          {/* Negotiation Coach */}
   319→          <Card>
   320→            <CardHeader>
   321→              <CardTitle className="flex items-center gap-2 text-lg">
   322→                <Calculator className="h-5 w-5" />
   323→                Outils d'aide
   324→              </CardTitle>
   325→            </CardHeader>
   326→            <CardContent className="space-y-3">
   327→              <NegotiationTrigger
   328→                propertyId={property.uniqueId}
   329→                propertyTitle={property.title}
   330→                variant="card"
   331→              />
   332→
   333→              <Link href="/finance" className="block">
   334→                <Card className="hover:bg-muted/50 cursor-pointer transition-colors">
   335→                  <CardContent className="p-4">
   336→                    <div className="flex items-center gap-3">
   337→                      <div className="rounded-full bg-green-100 p-2">
   338→                        <Calculator className="h-5 w-5 text-green-600" />
   339→                      </div>
   340→                      <div>
   341→                        <div className="font-medium">Simulateur de prêt</div>
   342→                        <div className="text-muted-foreground text-sm">
   343→                          Calculez vos mensualités
   344→                        </div>
   345→                      </div>
   346→                    </div>
   347→                  </CardContent>
   348→                </Card>
   349→              </Link>
   350→            </CardContent>
   351→          </Card>
   352→
   353→          {/* Energy info if available */}
   354→          {property.energyClass && (
   355→            <Card>
   356→              <CardHeader>
   357→                <CardTitle className="flex items-center gap-2 text-lg">
   358→                  <Zap className="h-5 w-5" />
   359→                  Performance énergétique
   360→                </CardTitle>
   361→              </CardHeader>
   362→              <CardContent>
   363→                <div className="flex items-center gap-4">
   364→                  <div
   365→                    className={`flex h-12 w-12 items-center justify-center rounded-lg text-lg font-bold text-white ${
   366→                      property.energyClass === "A"
   367→                        ? "bg-green-500"
   368→                        : property.energyClass === "B"
   369→                          ? "bg-lime-500"
   370→                          : property.energyClass === "C"
   371→                            ? "bg-yellow-500"
   372→                            : property.energyClass === "D"
   373→                              ? "bg-orange-400"
   374→                              : property.energyClass === "E"
   375→                                ? "bg-orange-500"
   376→                                : property.energyClass === "F"
   377→                                  ? "bg-red-500"
   378→                                  : "bg-red-700"
   379→                    }`}
   380→                  >
   381→                    {property.energyClass}
   382→                  </div>
   383→                  <div>
   384→                    <div className="font-medium">Classe DPE {property.energyClass}</div>
   385→                    {property.energyValue && (
   386→                      <div className="text-muted-foreground text-sm">
   387→                        {property.energyValue} kWh/m²/an
   388→                      </div>
   389→                    )}
   390→                  </div>
   391→                </div>
   392→                {property.gasClass && (
   393→                  <div className="mt-3 flex items-center gap-4">
   394→                    <div
   395→                      className={`flex h-12 w-12 items-center justify-center rounded-lg text-lg font-bold text-white ${
   396→                        property.gasClass === "A"
   397→                          ? "bg-green-500"
   398→                          : property.gasClass === "B"
   399→                            ? "bg-lime-500"
   400→                            : property.gasClass === "C"
   401→                              ? "bg-yellow-500"
   402→                              : property.gasClass === "D"
   403→                                ? "bg-orange-400"
   404→                                : property.gasClass === "E"
   405→                                  ? "bg-orange-500"
   406→                                  : property.gasClass === "F"
   407→                                    ? "bg-red-500"
   408→                                    : "bg-red-700"
   409→                      }`}
   410→                    >
   411→                      {property.gasClass}
   412→                    </div>
   413→                    <div>
   414→                      <div className="font-medium">Classe GES {property.gasClass}</div>
   415→                      {property.gasValue && (
   416→                        <div className="text-muted-foreground text-sm">
   417→                          {property.gasValue} kgCO2/m²/an
   418→                        </div>
   419→                      )}
   420→                    </div>
   421→                  </div>
   422→                )}
   423→              </CardContent>
   424→            </Card>
   425→          )}
   426→
   427→          {/* Contact info */}
   428→          {property.publisher && (
   429→            <Card>
   430→              <CardHeader>
   431→                <CardTitle className="text-lg">Contact</CardTitle>
   432→              </CardHeader>
   433→              <CardContent>
   434→                <div className="space-y-2">
   435→                  {property.publisher.name && (
   436→                    <div className="font-medium">{property.publisher.name}</div>
   437→                  )}
   438→                  <div className="text-muted-foreground text-sm capitalize">
   439→                    {property.publisher.type === "professional"
   440→                      ? "Professionnel"
   441→                      : "Particulier"}
   442→                  </div>
   443→                  {property.publisher.phone && (
   444→                    <Button variant="outline" size="sm" className="w-full" asChild>
   445→                      <a href={`tel:${property.publisher.phone}`}>
   446→                        Appeler: {property.publisher.phone}
   447→                      </a>
   448→                    </Button>
   449→                  )}
   450→                </div>
   451→              </CardContent>
   452→            </Card>
   453→          )}
   454→        </div>
   455→      </div>
   456→    </div>
   457→  );
   458→}
