    36→export function StatsSection() {

    39→      <div className="glass-card grid w-full items-center gap-8 p-8 sm:grid-cols-2 md:grid-cols-4 md:gap-0 lg:p-12">
    40→        {stats.map((stat, index) => (
    41→          <div
    42→            key={index}

    45→            {index > 0 && (
    46→              <div className="absolute left-0 top-1/2 hidden h-12 w-px -translate-y-1/2 bg-[var(--border-subtle)] md:block" />
    47→            )}
    48→            <h4 className="text-gradient-brand mb-2 text-4xl font-bold tabular-nums md:text-5xl">
    49→              <Counter from={0} to={stat.number} />
    50→              {stat.suffix}
    51→            </h4>
    52→            <p className="text-muted-foreground text-sm">{stat.text}</p>
    53→          </div>
    54→        ))}
    55→      </div>
    56→    </SectionLayout>
    57→  );
    58→}
    59→
    60→function Counter({
    61→  from,
    62→  to,
    63→  duration = 2,
    64→}: {
    65→  from: number;
    66→  to: number;
    67→  duration?: number;
    68→}) {
    69→  const nodeRef = useRef<HTMLSpanElement>(null);
    70→
    71→  useEffect(() => {
    72→    if (!nodeRef.current) return;
    73→    const node = nodeRef.current;
    74→
    75→    const controls = animate(from, to, {
    76→      duration,
    77→      ease: "easeInOut",
    78→
    79→      onUpdate(value) {
    80→        if (to >= 100) {
    81→          node.textContent = Math.round(value).toString();
    82→        } else {
    83→          node.textContent = value.toFixed(1);
    84→        }
    85→      },
    86→    });
    87→
    88→    return () => controls.stop();
    89→  }, [from, to, duration]);
    90→
    91→  return <span ref={nodeRef}>{from}</span>;
    92→}
