    69→export default async function AlbumDetailPage({ params, searchParams }: PageProps) {
    70→  const { id, locale } = await params;
    71→  const { preview } = await searchParams;
    72→  const isPreview = preview === "true";
    73→
    74→  // Si mode preview, vérifier l'authentification
    75→  let isAdmin = false;
    76→  if (isPreview) {
    77→    try {
    78→      const headersList = await headers();
    79→      const session = await auth.api.getSession({ headers: headersList });
    80→      isAdmin = session?.user?.role === "admin";
    81→    } catch {
    82→      isAdmin = false;
    83→    }
    84→
    85→    if (!isAdmin) {
    86→      notFound();
    87→    }
    88→  }
    89→
    90→  // Fetch the album
    91→  const album = await prisma.album.findUnique({
    92→    where: { id },
    93→  });
    94→
    95→  if (!album) {
    96→    notFound();
    97→  }
    98→
    99→  // Si pas en mode preview et album non publié, 404
   100→  if (!isPreview && !album.published) {
   101→    notFound();
   102→  }
   103→
   104→  // Fetch all albums for related section
   105→  const allAlbums = await prisma.album.findMany({
   106→    where: { published: true },
   107→    orderBy: { sortedDate: "desc" },
   108→  });
   109→
   110→  return (
   111→    <NeoAlbumDetail
   112→      album={album}
   113→      allAlbums={allAlbums}
   114→      locale={locale}
   115→      isPreview={isPreview && isAdmin}
   116→    />
   117→  );
   118→}
