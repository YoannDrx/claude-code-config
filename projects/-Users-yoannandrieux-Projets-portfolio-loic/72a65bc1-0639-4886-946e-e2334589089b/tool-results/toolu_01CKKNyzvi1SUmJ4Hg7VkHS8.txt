     1→"use client";
     2→
     3→import React from "react";

     5→import { ConsentGate } from "../legal/ConsentGate";
     6→
     7→interface NeoAlbumPlayerProps {
     8→  spotifyEmbed?: string | null;
     9→  youtubeEmbed?: string | null;
    10→  title: string;
    11→}
    12→
    13→// Extraire l'ID Spotify depuis différents formats d'URL
    14→const extractSpotifyId = (url: string): string | null => {
    15→  // Format embed: https://open.spotify.com/embed/album/xxx
    16→  // Format standard: https://open.spotify.com/album/xxx
    17→  const embedMatch = url.match(/embed\/album\/([a-zA-Z0-9]+)/);
    18→  if (embedMatch) return embedMatch[1];
    19→
    20→  const standardMatch = url.match(/album\/([a-zA-Z0-9]+)/);
    21→  if (standardMatch) return standardMatch[1];
    22→
    23→  // Si c'est déjà juste l'ID
    24→  if (/^[a-zA-Z0-9]{22}$/.test(url)) return url;
    25→
    26→  return null;
    27→};
    28→
    29→// Extraire l'ID YouTube depuis différents formats d'URL
    30→const extractYoutubeId = (url: string): string | null => {
    31→  // Format embed: https://www.youtube.com/embed/xxx
    32→  const embedMatch = url.match(/embed\/([a-zA-Z0-9_-]+)/);
    33→  if (embedMatch) return embedMatch[1];
    34→
    35→  // Format standard: https://www.youtube.com/watch?v=xxx
    36→  const watchMatch = url.match(/[?&]v=([a-zA-Z0-9_-]+)/);
    37→  if (watchMatch) return watchMatch[1];
    38→
    39→  // Format court: https://youtu.be/xxx
    40→  const shortMatch = url.match(/youtu\.be\/([a-zA-Z0-9_-]+)/);
    41→  if (shortMatch) return shortMatch[1];
    42→
    43→  // Si c'est déjà juste l'ID
    44→  if (/^[a-zA-Z0-9_-]{11}$/.test(url)) return url;
    45→
    46→  return null;
    47→};
